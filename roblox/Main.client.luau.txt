	end)

local function mk(parent, size, cf, color, mat, o)
	local p = Instance.new("Part")
	p.Anchored = true; p.CanCollide = false; p.CanTouch = false; p.CanQuery = false
	p.Size = size * S; p.CFrame = cf; p.Color = color
	p.Material = mat or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth; p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		if o.tr then p.Transparency = o.tr end
		if o.refl then p.Reflectance = o.refl end
		if o.shape then p.Shape = o.shape end
		if o.shadow == false then p.CastShadow = false end
	end
	p.Parent = parent
	return p
end

-- part welded to a car root (offsets in metres, optional X tilt in degrees)
local function cp(car, sx, sy, sz, ox, oy, oz, color, mat, o)
	local rot = CFrame.new()
	if o and o.rx then rot = CFrame.Angles(math.rad(o.rx), 0, 0) end
	local p = mk(car.model, Vector3.new(sx, sy, sz), car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * rot, color, mat, o)
	if color == car.paint and mat == nil then p.Reflectance = 0.12 end
	p.Anchored = false; p.Massless = true
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = p; wc.Parent = p
	return p
end

local GLASS = Color3.fromRGB(18, 28, 40)
-- sloped glass/panel welded to a car (tall end toward +Z by default; flip = tall end toward -Z)
local function cw(car, sx, sy, sz, ox, oy, oz, color, mat, flip, o)
	local w = Instance.new("WedgePart"); w.Anchored = false; w.CanCollide = false; w.CanTouch = false; w.CanQuery = false; w.Massless = true
	w.Size = Vector3.new(sx, sy, sz) * S; w.Color = color; w.Material = mat or Enum.Material.SmoothPlastic
	if o and o.refl then w.Reflectance = o.refl elseif color == car.paint and mat == nil then w.Reflectance = 0.12 end
	w.CFrame = car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * (flip and CFrame.Angles(0, math.pi, 0) or CFrame.new())
	w.Parent = car.model
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = w; wc.Parent = w
	return w
end

local function mirrors(car, x, y, z, c)

local function wheels(car, x, zf, zr, r, w)
	for _, sx in ipairs({ -x, x }) do
		for _, z in ipairs({ zf, zr }) do
			cp(car, w, r * 2, r * 2, sx, r, z, Color3.fromRGB(22, 22, 24), Enum.Material.Rubber, { shape = Enum.PartType.Cylinder })
			local rim = cp(car, w + 0.02, r * 1.25, r * 1.25, sx + (sx > 0 and 0.01 or -0.01), r, z, Color3.fromRGB(175, 180, 186), Enum.Material.Metal, { shape = Enum.PartType.Cylinder, refl = 0.25 })
			rim.Name = "Rim"
			cp(car, w + 0.04, r * 0.4, r * 0.4, sx + (sx > 0 and 0.02 or -0.02), r, z, Color3.fromRGB(60, 62, 66), Enum.Material.Metal, { shape = Enum.PartType.Cylinder })
		end
	end
end

local function lights(car, w, zF, zR, yH, yT, isPlayer)
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yH, zF, Color3.fromRGB(255, 250, 225), Enum.Material.Neon)
		local t = cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yT, zR, Color3.fromRGB(255, 30, 20), Enum.Material.Neon)

local function buildCar(kind, color)
	local car = { kind = kind, L = DIM[kind].L, W = DIM[kind].W, paint = color }
	car.model = Instance.new("Model"); car.model.Name = "Car_" .. kind
	car.root = mk(car.model, Vector3.new(0.4, 0.3, 0.4), CFrame.new(0, 300, 0), color, Enum.Material.SmoothPlastic, { tr = 1 })
	car.model.PrimaryPart = car.root
	BUILD[kind](car, color)
	car.model.Parent = world
	if kind == "vanta" or kind == "strada" or kind == "or1" then
		-- real headlight beams for the player's car
		local beam = Instance.new("SpotLight"); beam.Face = Enum.NormalId.Front; beam.Angle = 55; beam.Range = 60; beam.Brightness = 3
		beam.Color = Color3.fromRGB(255, 244, 220); beam.Shadows = true; beam.Parent = car.root
		local tail = Instance.new("PointLight"); tail.Range = 8; tail.Brightness = 1.2; tail.Color = Color3.fromRGB(255, 40, 30)
		local tp = car.model:FindFirstChild("Tail"); if tp then tail.Parent = tp end
	end
	return car
end

-- distant hills: huge squashed spheres that ride along with the camera so the horizon never runs out
local hills = {}
for i = 1, 16 do
	local p = Instance.new("Part"); p.Anchored = true; p.CanCollide = false; p.CanQuery = false; p.CanTouch = false; p.CastShadow = false
	p.Size = Vector3.new(4, 4, 4); p.Material = Enum.Material.Grass
	p.Color = Color3.fromRGB(70, 92, 62):Lerp(Color3.fromRGB(110, 120, 130), (i % 4) / 6)
	local mesh = Instance.new("SpecialMesh"); mesh.MeshType = Enum.MeshType.Sphere
	local w = rng:NextNumber(180, 340)
	mesh.Scale = Vector3.new(w, rng:NextNumber(45, 110), w * rng:NextNumber(0.6, 1)) * S / 4
	mesh.Parent = p
	p.Parent = world
	local side = i % 2 == 0 and -1 or 1
	hills[i] = { part = p, x = side * rng:NextNumber(260, 520), z = (i - 8) * 120 + rng:NextNumber(-40, 40) }
end
local function updateHills(f)
	for _, h in ipairs(hills) do
		local z = h.z - (f * 0.15) % 1920 -- slow parallax drift
		if z < -960 then z += 1920 end
		h.part.CFrame = CFrame.new(h.x * S, -6 * S, -(f + z + 700) * S)
	end
end
for i = 1, SEG_COUNT do segs[i] = { model = buildSegment(), k = i - 1 } end
local function placeSegment(s, k) s.k = k; s.model:PivotTo(CFrame.new(0, 0, -(k + 0.5) * SEG_LEN * S)) end
for i, s in ipairs(segs) do placeSegment(s, i - 4) end
local function updateRoad(f)
	updateHills(f)
	local cur = math.floor(f / SEG_LEN)
	for _, s in ipairs(segs) do
		if s.k < cur - 3 then placeSegment(s, s.k + SEG_COUNT) end
	end
end
local function setWet(on)
	for i, r in ipairs(roadParts) do
		r.Reflectance = on and 0.22 or 0
		local base = r.Name == "Road" and Color3.fromRGB(48, 49, 52) or Color3.fromRGB(62, 62, 64)
		r.Color = on and base:Lerp(Color3.new(0, 0, 0), 0.3) or base
	end
end

local function spawnTraffic(f, lane)
	local r = math.random()
	local kind = r < 0.5 and "sedan" or r < 0.75 and "suv" or r < 0.88 and "van" or "semi"
	lane = lane or math.random(1, 4)
	for _, c in ipairs(traffic) do
		if c.x == laneX(lane) and math.abs(c.f - f) < 34 then return nil end
		if c.change and c.change.to == lane and math.abs(c.f - f) < 34 then return nil end
	end
	local hue = math.random()
	local col = kind == "semi" and Color3.fromHSV(hue, 0.55, 0.7) or Color3.fromHSV(hue, math.random() < 0.4 and 0.05 or 0.6, math.random() < 0.4 and 0.35 or 0.8)
	local car = buildCar(kind, col)
	car.f = f; car.lane = lane; car.x = laneX(lane)
	car.desired = (kind == "semi" and rnd(17, 22) or rnd(19, 31)) + (5 - lane) * 1.4
	car.v = car.desired; car.rel = f - P.f; car.yawv = 0
	traffic[#traffic + 1] = car
	return car
end

local function clearTraffic()
	for _, c in ipairs(traffic) do c.model:Destroy() end
	traffic = {}
end

---------------------------------------------------------------- stunt run: ramps and nitro
local RAMP = { L = 11, W = 3.4, H = 1.7 }
local ramps, pickups = {}, {}
local stuntFolder = Instance.new("Folder"); stuntFolder.Name = "Stunt"; stuntFolder.Parent = world
local nextRampF, nextPickF = 0, 0
local function clearStunt()
	for _, r in ipairs(ramps) do r.model:Destroy() end
	for _, p in ipairs(pickups) do p.part:Destroy() end
	ramps, pickups = {}, {}
end
local function addPickup(f, x, y)
	local part = mk(stuntFolder, Vector3.new(0.9, 0.45, 0.45), CFrame.new(), Color3.fromRGB(70, 170, 255), Enum.Material.Neon, { shape = Enum.PartType.Cylinder, shadow = false })
	local glow = Instance.new("PointLight"); glow.Color = Color3.fromRGB(90, 180, 255); glow.Range = 10; glow.Brightness = 2; glow.Parent = part
	pickups[#pickups + 1] = { f = f, x = x, y = y, part = part }
end
local function addRamp(f)
	local lane = math.random(1, 4)
	local m = Instance.new("Model"); m.Name = "Ramp"
	local x = laneX(lane)
	-- WedgePart: tall end turned to face down the road so cars drive up it
	local w = Instance.new("WedgePart"); w.Anchored = true; w.CanCollide = false; w.CanQuery = false
	w.Size = Vector3.new(RAMP.W, RAMP.H, RAMP.L) * S; w.Color = Color3.fromRGB(245, 181, 36); w.Material = Enum.Material.DiamondPlate
	w.CFrame = CFrame.new(w2v(x, RAMP.H / 2, f)) * CFrame.Angles(0, math.pi, 0); w.Parent = m
	for _, sx in ipairs({ -1, 1 }) do
		mk(m, Vector3.new(0.2, 0.25, RAMP.L), CFrame.new(w2v(x + sx * (RAMP.W / 2 + 0.1), 0.12, f)), Color3.fromRGB(20, 20, 22), nil, { shadow = false })
	end
	local arrow = mk(m, Vector3.new(1.4, 0.03, 1.4), CFrame.new(w2v(x, 0.02, f + RAMP.L / 2 + 18)) * CFrame.Angles(0, math.rad(45), 0), Color3.fromRGB(245, 181, 36), Enum.Material.Neon, { shadow = false })
	arrow.Name = "Marker"
	m.Parent = stuntFolder
	ramps[#ramps + 1] = { f = f, x = x, lane = lane, model = m }
	-- keep the approach clear and hang canisters along the jump arc
	for _, c in ipairs(traffic) do
		if c.lane == lane and c.f > f - 40 and c.f < f + RAMP.L then c.f = f + RAMP.L + rnd(25, 45) end
	end
	for k = 1, 3 do addPickup(f + RAMP.L / 2 + 12 + k * 9, x, 3.2 + k * 0.4) end
end
local function stuntLand()
	local dist = P.f - P.airF
	P.jumps += 1; P.longest = math.max(P.longest, dist)
	local pts = math.floor(dist * 6 + P.cleared * 400)
	P.score += pts
	say(("JUMP %d m%s  +%d"):format(math.floor(dist), P.cleared > 0 and ("  ·  cleared %d"):format(P.cleared) or "", pts), Color3.fromRGB(140, 210, 255))
	play("land", 0.6, 0.7); shake = math.max(shake, 0.5)
	P.cleared = 0
end
local function updateStunt(dt)
	if mode ~= "stunt" then return end
	-- spawn ramps and loose canisters ahead
	if nextRampF < P.f + 120 then nextRampF = math.max(nextRampF, P.f + 220); addRamp(nextRampF); nextRampF += rnd(420, 700) end
	if nextPickF < P.f + 150 then nextPickF = math.max(nextPickF, P.f + 180); addPickup(nextPickF, laneX(math.random(1, 4)), 0.7); nextPickF += rnd(120, 220) end
	for i = #ramps, 1, -1 do if ramps[i].f < P.f - 120 then ramps[i].model:Destroy(); table.remove(ramps, i) end end
	-- ramp surface / launch
	local ground = 0
	for _, r in ipairs(ramps) do
		local t = (P.f - (r.f - RAMP.L / 2)) / RAMP.L
		if math.abs(P.x - r.x) < RAMP.W / 2 + 0.3 and t >= 0 and t <= 1 then
			ground = t * RAMP.H
			if not P.air and t > 0.97 then
				P.air = true; P.airF = P.f; P.cleared = 0; P.y = RAMP.H
				P.vy = P.v * (RAMP.H / RAMP.L) * 1.05 + 1.5
			end
		elseif not P.air and t >= 0 and t <= 1 and math.abs(P.x - r.x) < RAMP.W / 2 + 1.2 then
			P.v = P.v * (1 - dt * 0.8); P.damage += dt * 0.12 -- clipped the ramp's side
		end
	end
	if P.air then
		P.vy -= 15 * dt; P.y += P.vy * dt
		if P.y <= 0 then P.y = 0; P.vy = 0; P.air = false; stuntLand() end
	else
		P.y = ground
	end
	-- canisters
	for i = #pickups, 1, -1 do
		local pk = pickups[i]
		if pk.f < P.f - 30 then pk.part:Destroy(); table.remove(pickups, i)
		elseif math.abs(pk.f - P.f) < 2.6 and math.abs(pk.x - P.x) < 1.5 and math.abs(pk.y - (P.y + 0.6)) < 1.7 then
			P.nitro = math.min(1, P.nitro + 0.34); P.score += 250
			say("NITRO +250", Color3.fromRGB(110, 190, 255)); play("pickup", 0.4, 1.4)
			pk.part:Destroy(); table.remove(pickups, i)
		else
			pk.part.CFrame = CFrame.new(w2v(pk.x, pk.y + math.sin(os.clock() * 3 + i) * 0.1, pk.f)) * CFrame.Angles(0, os.clock() * 2, math.rad(90))
		end
	end
end

function startGame()
	clearTraffic()
	P = newP(); local c = CARS[sel]; P.top = c.top * MPH; P.acc = c.acc
	P.v = 12
	if not playerCar or playerCar.kind ~= c.id then showCar() end
	for _ = 1, 60 do spawnTraffic(rnd(45, 430)) end
	clearStunt(); nextRampF = 260; nextPickF = 120
	hist = {}; histT = 0; rewinding = nil; ind = 0; camPos = nil
	state = "play"; menu.Visible = false; over.Visible = false; hud.Visible = true
	say("GO!", Color3.fromRGB(140, 255, 160))
end
local function toMenu()
	state = "menu"; clearTraffic(); P = newP(); showCar(); menu.Visible = true; over.Visible = false; hud.Visible = false

local function wreck(why)
	if P.crashed then return end
	P.crashed = true; P.v = P.v * 0.6; shake = 1.5; play("crash", 0.9, 0.8)
	local e = Instance.new("Explosion"); e.BlastPressure = 0; e.BlastRadius = 1; e.DestroyJointRadiusPercent = 0
	e.Position = playerCar.root.Position + Vector3.new(0, 3, 0); e.Parent = Workspace
	P.crashT = 0
	overT.Text = why or "WRECKED"
end
local function finishGame()
	state = "over"
	local sc = math.floor(P.score)
	local newBest = sc > (bestBy[mode] or 0)
	bestBy[mode] = math.max(bestBy[mode] or 0, sc); best = bestBy[mode]
	submitBest(mode, sc)
	overS.Text = ("Score %s    %s %s    Distance %.2f mi"):format(tostring(sc), newBest and "NEW BEST!" or "Best", tostring(best), P.f / 1609.34)
	if mode == "stunt" then overS.Text ..= ("    Jumps %d · longest %d m"):format(P.jumps, math.floor(P.longest)) end
	over.Visible = true
end

end

---------------------------------------------------------------- simulation
local function stepPlayer(dt)
	local wet = rainOn and 0.85 or 1
	local throttle = (down(Enum.KeyCode.W, Enum.KeyCode.Up) and 1 or 0)
	local brake = (down(Enum.KeyCode.S, Enum.KeyCode.Down) and 1 or 0)
	local str = (down(Enum.KeyCode.D, Enum.KeyCode.Right) and 1 or 0) - (down(Enum.KeyCode.A, Enum.KeyCode.Left) and 1 or 0)
	if P.crashed then throttle, brake, str = 0, 1, 0 end
	P.steer = P.steer + clamp(str - P.steer, -dt * 5, dt * 5)
	local drag = 0.9 + 0.00045 * P.v * P.v
	P.boosting = mode == "stunt" and not P.crashed and P.nitro > 0 and down(Enum.KeyCode.LeftShift, Enum.KeyCode.N) ~= nil
	if P.boosting then P.nitro = math.max(0, P.nitro - dt * 0.2) end
	local top = P.boosting and P.top * 1.22 or P.top
	local a = throttle * P.acc * math.max(0, 1 - P.v / top) ^ 0.7 - drag - brake * 26 * wet * (P.crashed and 0.6 or 1)
	if P.boosting then a += 15 * math.max(0, 1 - P.v / top) end
	if P.air then a = -drag * 0.4 end -- no drive or brakes in the air
	P.v = math.max(0, P.v + a * dt)
	-- steering turns the car; with auto-straighten off the car keeps the heading you leave it at
	local rate = 1.35 / (1 + P.v / 42) * wet * (P.air and 0.15 or 1)
	P.heading = P.heading + P.steer * rate * dt
	if (autoCenter and str == 0) or P.crashed then P.heading = P.heading - P.heading * math.min(1, dt * 3.5) end
	P.heading = clamp(P.heading, -0.45, 0.45)
	P.vx = math.sin(P.heading) * P.v
	P.x = P.x + P.vx * dt
	P.yaw = -P.heading
	P.f = P.f + P.v * dt
	local edge = RAIL_X - 0.15 - (playerCar and playerCar.W / 2 or 1)
	if math.abs(P.x) > edge then
		P.x = clamp(P.x, -edge, edge); P.vx = 0; P.heading = -P.heading * 0.25 -- glance off the guardrail
		P.damage = P.damage + dt * 0.18; P.v = P.v * (1 - dt * 0.6)
		if P.damage >= 1 and not P.crashed then wreck("WRECKED") end
	end
end

local function stepTraffic(dt)
	for i = #traffic, 1, -1 do
		local c = traffic[i]
		-- car following
		local target = c.desired
		for _, o in ipairs(traffic) do
			if o ~= c and math.abs(o.x - c.x) < 2.4 and o.f > c.f then
				local gap = o.f - c.f - (o.L + c.L) / 2
				if gap < 8 + c.v * 1.1 then target = math.min(target, o.v - (8 + c.v - gap) * 0.05) end
			end
		end
		if c.yb then target = math.min(target, math.max(0, P.v * 0.72)) end
		c.v = c.v + clamp(target - c.v, -9 * dt, 3 * dt)
		c.f = c.f + c.v * dt
		-- lane change progress
		if c.change then
			c.change.t = c.change.t + dt / (c.change.fast and 1.8 or 2.8)
			local t = clamp(c.change.t, 0, 1)
			local e = t * t * (3 - 2 * t)
			local nx = laneX(c.change.from) + (laneX(c.change.to) - laneX(c.change.from)) * e
			c.yawv = -(nx - c.x) / math.max(dt, 1e-3) / math.max(c.v, 5)
			c.x = nx
			if c.change.t >= 1 then c.lane = c.change.to; c.x = laneX(c.lane); c.change = nil; c.yawv = 0 end
		end
		-- cull / respawn
		if c.f < P.f - 160 or c.f > P.f + 520 then
			c.model:Destroy(); table.remove(traffic, i)
		end
	end
	-- keep the road busy
	local want = 30
	local tries = 0
	while #traffic < want and tries < 6 do spawnTraffic(P.f + rnd(260, 440)); tries += 1 end
end

local function yieldToIndicator(dt)
	if ind == 0 or state ~= "play" or P.crashed then return end
	indT += dt
	local pl, bd = 2, 99
	for i = 1, 4 do local d = math.abs(P.x - laneX(i)); if d < bd then bd = d; pl = i end end
	if bd > 2.6 then return end
	local tl = pl + ind
	if tl < 1 or tl > 4 then return end
	for _, c in ipairs(traffic) do
		if c.lane == tl and not c.change and c.f - P.f > -30 and c.f - P.f < 60 then
			c.yr = c.yr or (indT + rnd(0.4, 1.3))
			if indT >= c.yr then
				local fl = tl + ind
				local free = fl >= 1 and fl <= 4
				if free then
					for _, o in ipairs(traffic) do
						if o ~= c and (o.lane == fl or (o.change and o.change.to == fl)) and math.abs(o.f - c.f) < 12 + c.v * 0.5 then free = false; break end
					end
				end
				if free then c.change = { from = c.lane, to = fl, t = -0.25, fast = true }; c.yb = false
				else c.yb = c.f - P.f < 8 end
			end
		elseif c.yb and c.lane ~= tl then c.yb = false end
	end
end

local function collisions(dt)
	if P.crashed then return end
	P.hitCd = math.max(0, P.hitCd - dt)
	for _, c in ipairs(traffic) do
		local df = c.f - P.f
		-- airborne over traffic (stunt run): no contact, count the cars cleared
		local clearH = c.kind == "semi" and 4.1 or (c.kind == "van" and 2.3 or 1.55)
		if P.air and P.y > clearH then
			if c.rel ~= nil and c.rel > 0 and df <= 0 and math.abs(c.x - P.x) < 2.5 then P.cleared += 1 end
			c.rel = df
			continue
		end
		local reach = (c.L + playerCar.L) / 2
		local dx = c.x - P.x
		local wr = (c.W + playerCar.W) / 2
		if math.abs(df) < reach and math.abs(dx) < wr then
			local overlap = wr - math.abs(dx)
			if overlap < 0.55 then
				-- side swipe
				local push = (dx >= 0 and -1 or 1)
				P.x = P.x + push * overlap * 0.5; P.heading = push * 0.06
				P.v = P.v * (1 - dt * 0.5)
				P.damage = P.damage + dt * 0.55; P.combo = 1; shake = math.max(shake, 0.4)
				if P.damage >= 1 then wreck("WRECKED") end
			elseif P.hitCd <= 0 then
				-- bumper hit: damage grows with the closing speed; a big one is a total wreck
				local closing = math.abs(P.v - c.v)
				local dmg = clamp(closing / 22, 0.2, 1.2)
				P.damage = P.damage + dmg; P.combo = 1; P.hitCd = 0.6; shake = math.max(shake, 0.6 + dmg)
				play("crash", 0.4 + 0.4 * math.min(1, dmg), 1)
				if df > 0 then P.v = math.min(P.v, c.v * 0.85); P.f = c.f - reach - 0.05
				else c.v = math.max(c.v, P.v + 2); c.f = P.f - reach - 0.05 end
				if P.damage >= 1 then wreck(closing > 25 and "TOTALLED" or "WRECKED")
				else say(("BUMP  -%d%%"):format(math.floor(dmg * 100)), Color3.fromRGB(255, 140, 120)) end
			end
			return
		end
		-- close calls
		local rel = df
		if c.rel ~= nil and ((c.rel > 0 and rel <= 0) or (c.rel < 0 and rel >= 0)) and math.abs(dx) < LANE_W * 1.1 and math.abs(dx) > (c.W + playerCar.W) / 2 - 0.05 and P.v > 15 then
			P.comboT = 3; P.combo = math.min(8, P.combo + 1); P.near += 1
			local pts = 50 * P.combo
			P.score += pts; say(("CLOSE CALL +%d  x%d"):format(pts, P.combo), Color3.fromRGB(255, 243, 176))
			play("closeCall", 0.35, 0.9 + P.combo * 0.06)
		end
		c.rel = rel
	end
end

local function blinkNow() return (os.clock() % 0.7) < 0.38 end

local lastBlink = false
local function updateIndicators()
	local on = blinkNow()
	if ind ~= 0 and on ~= lastBlink and state == "play" then play("tick", 0.25, on and 1.1 or 0.9) end
	lastBlink = on
	local function set(parts, lit) if parts then for _, p in ipairs(parts) do p.Color = lit and AMBER_ON or AMBER_OFF end end end
	set(playerCar.sigL, ind == -1 and on); set(playerCar.sigR, ind == 1 and on)
	indL.TextColor3 = (ind == -1 and on) and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(60, 50, 30)
	indR.TextColor3 = (ind == 1 and on) and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(60, 50, 30)
	for _, c in ipairs(traffic) do
		if c.change then
			local d = c.change.to - c.change.from
			if not c.sigL then
				-- traffic cars share the same builder; they only blink while changing lanes
			end
			if c.sigL then set(c.sigL, d < 0 and on); set(c.sigR, d > 0 and on) end
		elseif c.sigL then set(c.sigL, false); set(c.sigR, false) end
	end
end

local function updateCamera(dt)
	local car = playerCar
	local pos = car.root.Position
	local speedFov = clamp(P.v * 0.2, 0, 30)
	if camMode == 0 or state == "menu" then
		if state == "menu" then
			local a = os.clock() * 0.35
			camera.CFrame = CFrame.lookAt(pos + Vector3.new(math.sin(a) * 26, 8, math.cos(a) * 26), pos + Vector3.new(0, 2, 0))
			camera.FieldOfView = 55; setCarVisible(car, true)
			if cockpit.folder then cockpit.folder.Parent = nil end
		else
			local back = (16 + P.v * 0.05) * 1
			local want = pos + Vector3.new(P.x * 0.12 * S * -0.0 , 6.5, back * 1.0) + Vector3.new(-P.vx * 0.9, 0, 0)
			camPos = camPos and camPos:Lerp(want, math.min(1, dt * 7)) or want
			local look = pos + Vector3.new(0, 2.5, -22)
			local c = CFrame.lookAt(camPos, look)
			if shake > 0 then c = c * CFrame.new((math.random() - 0.5) * shake, (math.random() - 0.5) * shake, 0); shake = math.max(0, shake - dt * 2) end
			camera.CFrame = c
			camera.FieldOfView = 66 + speedFov + (P.boosting and 7 or 0)
			setCarVisible(car, true)
			if cockpit.folder then cockpit.folder.Parent = nil end
		end
	else
		local eye = car.root.CFrame * CFrame.new(-0.4 * S, 1.2 * S, -0.1 * S)
		local c = eye
		if shake > 0 then c = c * CFrame.new((math.random() - 0.5) * shake * 0.5, (math.random() - 0.5) * shake * 0.5, 0); shake = math.max(0, shake - dt * 2) end
		camera.CFrame = c; camera.FieldOfView = 74 + speedFov * 0.6
		setCarVisible(car, false)
		layoutCockpit(c)
		cockpit.folder.Parent = world
	end
end

local function updateHud(dt)
	scoreL.Text = tostring(math.floor(P.score))
	distL.Text = ("%.2f mi"):format(P.f / 1609.34)
	speedL.Text = tostring(math.floor(P.v / MPH + 0.5))
	dmgBar.Size = UDim2.fromScale(clamp(P.damage, 0, 1), 1)
	comboL.Text = P.combo > 1 and ("COMBO x%d"):format(P.combo) or ""
	nitroBack.Visible = mode == "stunt"; nitroL.Visible = mode == "stunt"
	nitroBar.Size = UDim2.fromScale(clamp(P.nitro, 0, 1), 1)
	nitroBar.BackgroundColor3 = P.boosting and Color3.fromRGB(150, 220, 255) or Color3.fromRGB(70, 170, 255)
	airL.Visible = P.air; if P.air then airL.Text = ("AIR  %d m"):format(math.floor(P.f - P.airF)) end
	if toastT > 0 then toastT -= dt; toast.TextTransparency = clamp(1 - toastT * 0.2, 0, 1) * 0 + (toastT < 0.4 and (1 - toastT / 0.4) or 0) else toast.TextTransparency = 1 end
end

local nextHist = 0
RunService.RenderStepped:Connect(function(dtRaw)
	local dt = math.min(dtRaw, 0.05)
	camera.CameraType = Enum.CameraType.Scriptable
	if state == "menu" then
		if not playerCar then showCar(); refreshMenu() end
		poseCar(playerCar, laneX(2), 0, 0, 0)
		updateRoad(0)
		updateCamera(dt)
		rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 40, 0))
		rain.Rate = rainOn and 1400 or 0
		return
	end
	if state == "paused" then return end

	if rewinding then
		-- scrub backwards through the recorded snapshots (about a second for 7 s of history)
		rewinding.acc += dt * (#hist / 1.0)
		local steps = math.floor(rewinding.acc); rewinding.acc -= steps
		rewinding.i = math.max(1, rewinding.i - steps)
		restore(hist[rewinding.i])
		rewTint.BackgroundTransparency = 0.82
		if rewinding.i <= 1 then
			for k = #hist, rewinding.i + 1, -1 do hist[k] = nil end
			rewinding = nil; rewLbl.Visible = false; rewTint.BackgroundTransparency = 1; P.crashed = false
			say("REWIND", Color3.fromRGB(190, 225, 255))
		end
	else
		if P.crashed then
			P.crashT += dt; stepPlayer(dt); stepTraffic(dt)
			if P.crashT > 1.4 and state == "play" then finishGame() end
		else
			stepPlayer(dt); updateStunt(dt); yieldToIndicator(dt); stepTraffic(dt); collisions(dt)
			P.score += P.v * dt * 0.1
			if P.comboT > 0 then P.comboT -= dt; if P.comboT <= 0 then P.combo = 1 end end
			-- history every 0.1 s
			histT += dt
			if histT >= 0.1 then
				histT -= 0.1; hist[#hist + 1] = snapshot()
				if #hist > REWIND_S * 10 + 1 then table.remove(hist, 1) end
			end
		end
	end

	updateRoad(P.f)
	poseCar(playerCar, P.x, P.f, P.yaw, P.crashed and math.min(0.6, (P.crashT or 0) * 0.5) or 0, P.y)
	if P.air or P.y > 0.01 then playerCar.root.CFrame *= CFrame.Angles(clamp(P.vy * 0.03, -0.3, 0.3), 0, 0) end
	for _, c in ipairs(traffic) do poseCar(c, c.x, c.f, c.yawv or 0, 0) end
	updateIndicators()
	updateCamera(dt)
	updateHud(dt)
	rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 26, -12)) * CFrame.Angles(-math.atan2(P.v * S, 150) * 0.6, 0, 0)
	rain.Rate = rainOn and 1400 or 0
	rain.Speed = NumberRange.new(140, 170)
	local headCar = playerCar
end)
