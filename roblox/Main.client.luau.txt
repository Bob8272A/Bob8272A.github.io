-- Web layout: our carriageway is x > 0, the oncoming one is mirrored at x < 0
local LANES = { 3.65, 7.35, 11.05, 14.75 }
local function laneX(i) return LANES[i] end
local OUTER_WALL, INNER_WALL = 20.08, 0.32
local ROAD_X1 = 19.6
local TERRAIN = { amp = 0, bend = 0 }
local function hillH(s) return math.sin(s * 0.0041 + 1.3) + math.sin(s * 0.0093 + 0.4) * 0.55 + math.sin(s * 0.021 + 2.1) * 0.18 end
local function hillX(s) return math.sin(s * 0.0023 + 0.7) + math.sin(s * 0.0061 + 2.9) * 0.45 end
local function roadOff(f) return hillX(f) * TERRAIN.bend, hillH(f) * TERRAIN.amp end
local function rnd(a, b) return a + math.random() * (b - a) end
local function clamp(v, a, b) return math.max(a, math.min(b, v)) end
-- road coordinates (x across, y up, f along) -> world studs, following hills and curves
local function w2v(x, y, f)
	local ox, oy = roadOff(f)
	return Vector3.new((x + ox) * S, (y + oy) * S, -f * S)
end
-- a frame on the road surface at (x, y, f) that faces down the road
local function roadCF(x, y, f)
	local p = w2v(x, y, f)
	if TERRAIN.amp == 0 and TERRAIN.bend == 0 then return CFrame.new(p) end
	local x1, y1 = roadOff(f + 1)
	local x0, y0 = roadOff(f - 1)
	return CFrame.lookAt(p, p + Vector3.new((x1 - x0) / 2, (y1 - y0) / 2, -1))
end

local function mk(parent, size, cf, color, mat, o)
	local p = Instance.new("Part")
	p.Anchored = true; p.CanCollide = false; p.CanTouch = false; p.CanQuery = false
	p.Size = size * S; p.CFrame = cf; p.Color = color
	p.Material = mat or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth; p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		if o.tr then p.Transparency = o.tr end
		if o.refl then p.Reflectance = o.refl end
		if o.shape then p.Shape = o.shape end
		if o.shadow == false then p.CastShadow = false end
	end
	p.Parent = parent
	return p
end

---------------------------------------------------------------- car builder: every car is assembled from a spec
-- Units are metres, the car faces -Z, y = 0 is the road. Each part is welded to an invisible root.
local GLASS = Color3.fromRGB(16, 24, 34)
local BLACK = Color3.fromRGB(14, 15, 17)
local TRIM = Color3.fromRGB(28, 29, 32)
local CHROME = Color3.fromRGB(196, 200, 206)
local RUBBER = Color3.fromRGB(20, 20, 22)
local LED = Color3.fromRGB(255, 250, 232)
local TAIL = Color3.fromRGB(255, 26, 18)
local ROT_Z = CFrame.Angles(0, math.pi / 2, 0) -- turns a cylinder's axis from X to Z

local function attach(car, p)
	p.Anchored = false; p.Massless = true; p.CanCollide = false; p.CanTouch = false; p.CanQuery = false
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = p; wc.Parent = p
	p.Parent = car.model
	return p
end
-- box/cylinder/ball part: size (sx, sy, sz) at (ox, oy, oz), optional extra rotation
local function cp(car, sx, sy, sz, ox, oy, oz, color, mat, o)
	local p = Instance.new("Part")
	p.Size = Vector3.new(math.max(sx, 0.02), math.max(sy, 0.02), math.max(sz, 0.02)) * S
	p.Color = color; p.Material = mat or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth; p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		if o.shape then p.Shape = o.shape end
		if o.refl then p.Reflectance = o.refl end
		if o.tr then p.Transparency = o.tr end
		if o.shadow == false then p.CastShadow = false end
	end
	if color == car.paint and (mat == nil or mat == Enum.Material.SmoothPlastic) and not (o and o.refl) then p.Reflectance = 0.14 end
	local rot = o and o.rot or CFrame.new()
	if o and o.rx then rot = rot * CFrame.Angles(math.rad(o.rx), 0, 0) end
	p.CFrame = car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * rot
	return attach(car, p)
end
-- sloped panel: tall end toward +Z (rear) unless flip
local function cw(car, sx, sy, sz, ox, oy, oz, color, mat, flip, o)
	local w = Instance.new("WedgePart")
	w.Size = Vector3.new(sx, sy, sz) * S; w.Color = color; w.Material = mat or Enum.Material.SmoothPlastic
	if o and o.refl then w.Reflectance = o.refl elseif color == car.paint and mat == nil then w.Reflectance = 0.14 end
	w.CFrame = car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * (flip and CFrame.Angles(0, math.pi, 0) or CFrame.new())
	return attach(car, w)
end
-- box with rounded long edges: two crossed boxes plus four cylinders along the length
local function rbox(car, w, h, l, x, y, z, r, color, mat)
	if not car.hi then cp(car, w, h, l, x, y, z, color, mat); return end
	r = math.min(r, h / 2 - 0.01, w / 2 - 0.01)
	cp(car, w - 2 * r, h, l, x, y, z, color, mat)
	cp(car, w, h - 2 * r, l, x, y, z, color, mat)
	for _, sx in ipairs({ -1, 1 }) do
		for _, sy in ipairs({ -1, 1 }) do
			cp(car, l, 2 * r, 2 * r, x + sx * (w / 2 - r), y + sy * (h / 2 - r), z, color, mat, { shape = Enum.PartType.Cylinder, rot = ROT_Z })
		end
	end
end
local function wheel(car, x, z, r, w, rimCol, caliper, spokes)
	local side = x > 0 and 1 or -1
	cp(car, w, r * 2, r * 2, x, r, z, RUBBER, Enum.Material.Rubber, { shape = Enum.PartType.Cylinder })
	if not car.hi then
		cp(car, 0.04, r * 1.3, r * 1.3, x + side * (w / 2 + 0.006), r, z, rimCol, Enum.Material.Metal, { shape = Enum.PartType.Cylinder, refl = 0.2 })
		return
	end
	cp(car, w * 0.5, r * 1.92, r * 1.92, x + side * w * 0.3, r, z, Color3.fromRGB(34, 34, 37), Enum.Material.Rubber, { shape = Enum.PartType.Cylinder }) -- sidewall
	cp(car, 0.03, r * 1.34, r * 1.34, x + side * (w / 2 + 0.006), r, z, Color3.fromRGB(40, 41, 44), Enum.Material.Metal, { shape = Enum.PartType.Cylinder }) -- disc behind spokes
	if caliper then cp(car, 0.05, r * 0.42, r * 0.3, x + side * (w / 2 - 0.02), r + r * 0.35, z + r * 0.25, caliper, Enum.Material.SmoothPlastic) end
	local n = spokes or 5
	for k = 0, n - 1 do
		local a = k * 2 * math.pi / n
		cp(car, 0.035, r * 1.28, r * 0.15, x + side * (w / 2 + 0.02), r, z, rimCol, Enum.Material.Metal, { rot = CFrame.Angles(a, 0, 0), refl = 0.3 })
	end
	cp(car, 0.04, r * 1.36, r * 1.36, x + side * (w / 2 + 0.012), r, z, rimCol, Enum.Material.Metal, { shape = Enum.PartType.Cylinder, refl = 0.3, tr = 0.0 }).Transparency = 0.82 -- rim lip
	cp(car, 0.05, r * 0.36, r * 0.36, x + side * (w / 2 + 0.035), r, z, rimCol, Enum.Material.Metal, { shape = Enum.PartType.Cylinder, refl = 0.35 }) -- hub
end
local function signals(car, x, yF, zF, yR, zR)
	for _, sx in ipairs({ -1, 1 }) do
		local f = cp(car, 0.16, 0.07, 0.04, sx * x, yF, zF, AMBER_OFF, Enum.Material.Neon, { shadow = false })
		local r = cp(car, 0.16, 0.07, 0.04, sx * x, yR, zR, AMBER_OFF, Enum.Material.Neon, { shadow = false })
		if sx < 0 then car.sigL = { f, r } else car.sigR = { f, r } end
	end
end
local function plate(car, y, z)
	local s = z > 0 and 1 or -1
	cp(car, 0.52, 0.12, 0.02, 0, y, z + s * 0.005, Color3.fromRGB(236, 236, 230), Enum.Material.SmoothPlastic)
	cp(car, 0.36, 0.04, 0.022, 0, y, z + s * 0.008, Color3.fromRGB(40, 60, 120), Enum.Material.SmoothPlastic)
end

-- generic passenger car body from a spec
local function body(car, sp, c)
	local L, W, top = sp.L, sp.W, sp.clear + sp.body
	local yb = sp.clear + sp.body / 2
	local zf, zr = -L / 2, L / 2
	-- lower body with rounded edges, and a darker lower sill
	rbox(car, W, sp.body, L - 0.24, 0, yb, 0, 0.12, c)
	rbox(car, W - 0.1, sp.body * 0.72, 0.3, 0, yb - sp.body * 0.12, zf + 0.15, 0.12, c) -- front bumper rounds
	rbox(car, W - 0.1, sp.body * 0.72, 0.3, 0, yb - sp.body * 0.12, zr - 0.15, 0.12, c) -- rear bumper rounds
	cp(car, W + 0.01, 0.1, L - 1.6, 0, sp.clear + 0.06, 0, sp.sill or TRIM) -- side skirts
	-- nose slope and tail slope
	cw(car, W - 0.06, sp.nose or 0.12, 0.55, 0, top + (sp.nose or 0.12) / 2, zf + 0.42, c, nil, false)
	-- wheel arches (dark discs on the flanks)
	for _, sx in ipairs({ -1, 1 }) do
		for _, z in ipairs({ -sp.wb / 2, sp.wb / 2 }) do
			cp(car, 0.04, (sp.wr + 0.08) * 2, (sp.wr + 0.08) * 2, sx * (W / 2 + 0.003), sp.wr, z, Color3.fromRGB(10, 10, 11), Enum.Material.SmoothPlastic, { shape = Enum.PartType.Cylinder })
		end
	end
	-- greenhouse: windscreen, side glass, roof, rear glass
	local cw0 = W - (sp.tumble or 0.22) * 2
	local zWs = zf + sp.hood -- base of the windscreen
	local zRoof0 = zWs + sp.ws
	local zRoof1 = zRoof0 + sp.cabin
	local yG = top + sp.roof / 2
	cw(car, cw0, sp.roof, sp.ws, 0, yG, zWs + sp.ws / 2, GLASS, Enum.Material.Glass, false, { refl = 0.25 })
	cp(car, cw0, sp.roof, sp.cabin, 0, yG, (zRoof0 + zRoof1) / 2, GLASS, Enum.Material.Glass, { refl = 0.25 })
	if sp.rg > 0.05 then
		cw(car, cw0, sp.roof, sp.rg, 0, yG, zRoof1 + sp.rg / 2, sp.fastback and c or GLASS, sp.fastback and nil or Enum.Material.Glass, true, { refl = sp.fastback and 0.14 or 0.25 })
	end
	rbox(car, cw0 + 0.02, 0.07, sp.cabin + 0.06, 0, top + sp.roof + 0.03, (zRoof0 + zRoof1) / 2, 0.03, sp.roofCol or c)
	-- pillars and window frame lines
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, 0.04, sp.roof, 0.09, sx * (cw0 / 2 + 0.005), yG, zRoof0 + sp.cabin * 0.48, sp.roofCol or c) -- B-pillar
		cp(car, 0.035, 0.035, sp.cabin + sp.ws * 0.6, sx * (cw0 / 2 + 0.006), top + 0.02, (zRoof0 + zRoof1) / 2 - sp.ws * 0.2, CHROME, Enum.Material.Metal) -- belt line
		-- door seams and handles
		if car.hi then cp(car, 0.012, sp.body * 0.8, 0.012, sx * (W / 2 + 0.002), yb, zRoof0 - 0.15, Color3.fromRGB(10, 10, 10))
		cp(car, 0.012, sp.body * 0.8, 0.012, sx * (W / 2 + 0.002), yb, zRoof0 + sp.cabin * 0.48, Color3.fromRGB(10, 10, 10))
		cp(car, 0.02, 0.03, 0.14, sx * (W / 2 + 0.008), top - 0.12, zRoof0 + sp.cabin * 0.3, CHROME, Enum.Material.Metal) end
		-- mirrors
		cp(car, 0.18, 0.03, 0.05, sx * (cw0 / 2 + 0.12), top + 0.1, zWs + sp.ws * 0.35, BLACK)
		rbox(car, 0.17, 0.11, 0.14, sx * (cw0 / 2 + 0.25), top + 0.12, zWs + sp.ws * 0.35, 0.04, c)
	end
	-- front: grille, intake, splitter, headlights with LED strips
	local yF = sp.clear + sp.body * 0.62
	cp(car, W * 0.42, sp.body * 0.3, 0.04, 0, yF - 0.03, zf - 0.005, BLACK)
	for k = 1, 3 do cp(car, W * 0.4, 0.015, 0.045, 0, yF - 0.1 + k * sp.body * 0.075, zf - 0.008, sp.grilleTrim or CHROME, Enum.Material.Metal) end
	cp(car, W * 0.7, 0.09, 0.05, 0, sp.clear + 0.1, zf - 0.005, BLACK) -- lower intake
	if sp.splitter then cp(car, W - 0.1, 0.03, 0.18, 0, sp.clear + 0.02, zf - 0.04, Color3.fromRGB(22, 22, 24), Enum.Material.SmoothPlastic) end
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, W * 0.24, 0.12, 0.05, sx * W * 0.33, yF + 0.04, zf + 0.005, Color3.fromRGB(26, 30, 36), Enum.Material.Glass, { refl = 0.4 })
		cp(car, W * 0.22, 0.03, 0.055, sx * W * 0.33, yF + 0.08, zf, sp.led or LED, Enum.Material.Neon, { shadow = false })
		cp(car, 0.09, 0.07, 0.055, sx * W * 0.4, yF + 0.02, zf, LED, Enum.Material.Neon, { shadow = false })
	end
	-- rear: light bar, diffuser, exhausts, plate
	local yR = sp.clear + sp.body * 0.72
	if sp.lightbar ~= false then
		cp(car, W * 0.86, 0.045, 0.04, 0, yR + 0.04, zr + 0.005, TAIL, Enum.Material.Neon, { shadow = false }).Name = "Tail"
	end
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, W * 0.2, 0.11, 0.05, sx * W * 0.36, yR, zr + 0.004, Color3.fromRGB(90, 8, 6), Enum.Material.Glass, { refl = 0.3 })
		cp(car, W * 0.17, 0.05, 0.055, sx * W * 0.36, yR, zr + 0.003, TAIL, Enum.Material.Neon, { shadow = false })
	end
	cp(car, W * 0.62, 0.1, 0.06, 0, sp.clear + 0.06, zr - 0.01, Color3.fromRGB(20, 20, 22)) -- diffuser
	for _, sx in ipairs(sp.exhausts or { -1, 1 }) do
		cp(car, 0.16, 0.09, 0.09, sx * W * 0.3, sp.clear + 0.1, zr + 0.02, CHROME, Enum.Material.Metal, { shape = Enum.PartType.Cylinder, rot = ROT_Z, refl = 0.4 })
	end
	plate(car, sp.clear + 0.22, zr)
	plate(car, sp.clear + 0.16, zf)
	signals(car, W * 0.44, yF + 0.04, zf + 0.002, yR, zr + 0.004)
	-- wheels
	for _, sx in ipairs({ -1, 1 }) do
		for _, z in ipairs({ -sp.wb / 2, sp.wb / 2 }) do
			wheel(car, sx * (W / 2 - sp.ww / 2 + 0.02), z, sp.wr, sp.ww, sp.rim or Color3.fromRGB(176, 181, 187), sp.caliper, sp.spokes)
		end
	end
	return top, zWs, zRoof0, zRoof1
end

local function updateHills(f)
	for _, h in ipairs(hills) do
		local z = h.z - (f * 0.15) % 1920 -- slow parallax drift
		if z < -960 then z += 1920 end
		local ff = f + z + 700
		local ox, oy = roadOff(ff)
		h.part.CFrame = CFrame.new((h.x + ox) * S, (oy - 6) * S, -ff * S)
	end

function updateRoad(f)
	updateHills(f)
	local key = TERRAIN.amp .. "/" .. TERRAIN.bend
	if key ~= terrainKey then terrainKey = key; for _, s in ipairs(segs) do s.k = nil end end -- terrain changed: re-place everything
	local base = math.floor(f / SEG_LEN) - 3
	for i, s in ipairs(segs) do
		local k = base + ((i - 1 - base) % SEG_COUNT)
		if s.k ~= k then placeSegment(s, k) end
	end
end

-- traffic tuning follows the web version's endless curve: heavier and faster the farther you go
local function updateDiff()
	local D = curDiff()
	local amt = clamp(SET.traffic, 0, 2)
	if SET.mode == "rush" then -- dense, slow traffic
		local t = math.min(1, P.f / (6 * MILE))
		DIFF.traffic = math.floor((38 + t * 10) * D.traffic * amt + 0.5)
		DIFF.speed = (0.68 + t * 0.2) * D.speed
		DIFF.lane = math.min(0.9, (0.3 + t * 0.3) * D.lane)
	else
		local t = math.min(1, P.f / (8 * MILE))
		DIFF.traffic = math.floor((10 + t * 22) * 1.2 * D.traffic * amt + 0.5)
		DIFF.speed = (0.92 + t * 0.33) * D.speed
		DIFF.lane = math.min(0.9, (0.14 + t * 0.46) * D.lane)
	end
end

local function laneFree(lane, f, except, gap)
	for _, o in ipairs(traffic) do
		if o ~= except and (o.lane == lane or (o.change and o.change.to == lane)) and math.abs(o.f - f) < gap + (o.L + 4.6) / 2 then return false end
	end
	return true
end

local KINDS = { { "sedan", 40 }, { "suv", 22 }, { "hatch", 12 }, { "pickup", 9 }, { "semi", 11 }, { "van", 6 } }
local TPAINT = { { Color3.fromRGB(241, 241, 238), 24 }, { Color3.fromRGB(12, 13, 15), 19 }, { Color3.fromRGB(90, 94, 98), 17 }, { Color3.fromRGB(185, 189, 193), 14 },
	{ Color3.fromRGB(30, 60, 120), 9 }, { Color3.fromRGB(139, 16, 21), 8 }, { Color3.fromRGB(44, 74, 53), 3 }, { Color3.fromRGB(106, 74, 44), 3 }, { Color3.fromRGB(214, 205, 188), 3 } }
local function weighted(list) local r = math.random() * 100; for _, e in ipairs(list) do r -= e[2]; if r <= 0 then return e[1] end end return list[1][1] end
TRAFFIC_PAINT = function() return weighted(TPAINT) end
local function laneSpeed(lane, kind)
	if kind == "semi" then return rnd(23.5, 26.5) * math.min(DIFF.speed, 1.15) end
	local r = ({ { 33, 37.5 }, { 30, 34 }, { 27, 31 }, { 24.5, 28.5 } })[lane]
	return rnd(r[1], r[2]) * DIFF.speed
end
local function spawnTraffic(f, lane)
	local kind = weighted(KINDS)
	lane = lane or math.random(1, 4)
	if kind == "semi" and lane == 1 then lane = math.random(2, 4) end
	if not laneFree(lane, f, nil, SET.mode == "rush" and 9 or 30) then return nil end
	local col = kind == "semi" and Color3.fromHSV(math.random(), 0.55, 0.7) or TRAFFIC_PAINT()
	local car = buildCar(kind, col)
	car.f = f; car.lane = lane; car.x = laneX(lane)
	car.desired = laneSpeed(lane, kind)
	car.v = car.desired; car.yawv = 0; car.side = sgn(f - P.f); car.timer = rnd(2, 6)
	traffic[#traffic + 1] = car
	return car
end
local function clearTraffic()
	for _, c in ipairs(traffic) do c.model:Destroy() end
	traffic = {}

		fill.Size = UDim2.fromScale(t, 1); knob.Position = UDim2.new(0, 8 + t * (rail.AbsoluteSize.X), 0.5, 0)
	end

	task.defer(refresh)

---------------------------------------------------------------- snapshots / rewind
local function snapshot()
	local s = { p = {}, cars = {}, ind = ind, pur = table.clone(MD.PUR), rush = table.clone(MD.RUSH), cops = {} }
	for k, v in pairs(P) do s.p[k] = v end
	for i, c in ipairs(MD.cops) do s.cops[i] = { f = c.f, x = c.x, v = c.v, active = c.active, crashed = c.crashed, yaw = c.yaw } end
	for _, c in ipairs(traffic) do
		s.cars[#s.cars + 1] = { c = c, f = c.f, x = c.x, v = c.v, lane = c.lane, crashed = c.crashed, yawv = c.yawv }
	end
	return s
end
-- Put the run back to a snapshot. Cars that did not exist yet are removed for real (not just hidden),
-- so nothing invisible is left on the road to crash into.
local function restore(s)
	for k, v in pairs(s.p) do P[k] = v end
	ind = s.ind
	for k, v in pairs(s.pur) do MD.PUR[k] = v end
	for k, v in pairs(s.rush) do MD.RUSH[k] = v end
	for i, e in ipairs(s.cops) do
		local c = MD.cops[i]
		c.f = e.f; c.x = e.x; c.v = e.v; c.active = e.active; c.crashed = e.crashed; c.yaw = e.yaw; c.vx = 0
		c.car.model.Parent = e.active and world or nil
	end
	local keep, list = {}, {}
	for _, e in ipairs(s.cars) do
		local c = e.c
		if c.model.Parent then
			c.f = e.f; c.x = e.x; c.v = e.v; c.lane = e.lane; c.crashed = e.crashed; c.yawv = e.yawv or 0
			c.change = nil; c.yb = false; c.yr = nil; c.side = sgn(c.f - P.f); c.counted = false
			keep[c] = true; list[#list + 1] = c
		end
	end
	for _, c in ipairs(traffic) do if not keep[c] then c.model:Destroy() end end
	traffic = list
end

local function addRamp(f)

	w.CFrame = roadCF(x, RAMP.H / 2, f) * CFrame.Angles(0, math.pi, 0); w.Parent = m
	for _, sx in ipairs({ -1, 1 }) do
		mk(m, Vector3.new(0.2, 0.25, RAMP.L), roadCF(x + sx * (RAMP.W / 2 + 0.1), 0.12, f), Color3.fromRGB(20, 20, 22), nil, { shadow = false })
	end

	end
	m.Parent = stuntFolder
	ramps[#ramps + 1] = { f = f, x = x, model = m }
	for k = 1, 3 do addPickup(f + RAMP.L / 2 + 12 + k * 9, x - k * 1.6, 3.2 + k * 0.4) end
end
local function stuntLand()
	local dist = P.f - P.airF
	P.jumps += 1; P.longest = math.max(P.longest, dist)
	local pts = math.floor((dist * 6 + P.cleared * 400) * paceMult(P.fwd * 2.23694))
	P.score += pts
	popup(("BIG AIR · %d m"):format(math.floor(dist)), (P.cleared > 0 and ("Cleared %d · "):format(P.cleared) or "") .. "+" .. commas(pts))
	play("land", 0.6, 0.7); shake = math.max(shake, 0.5)
	P.cleared = 0
end
local function updateStunt(dt)
	if SET.mode ~= "stunt" then return end
	if nextRampF < P.f + 140 then nextRampF = math.max(nextRampF, P.f + 240); addRamp(nextRampF); nextRampF += rnd(420, 700) end
	if nextPickF < P.f + 150 then nextPickF = math.max(nextPickF, P.f + 180); addPickup(nextPickF, laneX(math.random(1, 4)), 0.7); nextPickF += rnd(120, 220) end
	for i = #ramps, 1, -1 do if ramps[i].f < P.f - 150 or ramps[i].f > P.f + 900 then ramps[i].model:Destroy(); table.remove(ramps, i) end end
	local ground, onRamp = 0, false
	for _, r in ipairs(ramps) do
		local t = (P.f - (r.f - RAMP.L / 2)) / RAMP.L
		local inLane = math.abs(P.x - r.x) < RAMP.W / 2 + 0.3
		if inLane and t >= 0 and t <= 1 then
			ground = t * RAMP.H; onRamp = true
		end
		-- launch when the car crosses the lip (checked by crossing, so a fast frame can't skip it)
		if inLane and not P.air and r.lastT and r.lastT >= 0 and r.lastT < 1 and t >= 0.97 then
			P.air = true; P.airF = P.f; P.cleared = 0; P.y = RAMP.H; onRamp = false
			P.vy = P.v * (RAMP.H / RAMP.L) * 1.05 + 1.5
		end
		r.lastT = t
	end
	if P.air then
		P.vy -= 15 * dt; P.y += P.vy * dt
		if P.y <= 0 then P.y = 0; P.vy = 0; P.air = false; stuntLand() end
	elseif onRamp then P.y = ground else P.y = 0 end
	for i = #pickups, 1, -1 do
		local pk = pickups[i]
		if pk.f < P.f - 30 or pk.f > P.f + 900 then pk.part:Destroy(); table.remove(pickups, i)
		elseif math.abs(pk.f - P.f) < 2.6 and math.abs(pk.x - P.x) < 1.5 and math.abs(pk.y - (P.y + 0.6)) < 1.7 then
			P.nitro = math.min(1, P.nitro + 0.34); P.score += 250
			popup("NITRO", "+250"); play("pickup", 0.4, 1.4)
			pk.part:Destroy(); table.remove(pickups, i)
		else
			pk.part.CFrame = CFrame.new(w2v(pk.x, pk.y + math.sin(os.clock() * 3 + i) * 0.1, pk.f)) * CFrame.Angles(0, os.clock() * 2, math.rad(90))
		end
	end
end

---------------------------------------------------------------- crash, damage and close calls (rules ported from the web version)
local KIND = { sedan = "a sedan", hatch = "a hatchback", suv = "an SUV", pickup = "a pickup", van = "a delivery van", semi = "a semi truck" }
local KIND_UP = { sedan = "SEDAN", hatch = "HATCHBACK", suv = "SUV", pickup = "PICKUP", van = "VAN", semi = "BIG RIG" }
local function cap(s) return s:sub(1, 1):upper() .. s:sub(2) end
function crash(msg, other, extra)
	if P.crashed then return end
	P.crashed = true; P.crashT = 0; P.msg = msg
	local E = clamp((P.fwd + (extra or 0) * 0.8 - 15) / 75, 0, 1); P.crashE = E
	local vlat = math.sin(P.h) * P.v
	P.cvz = P.fwd * (other and 0.55 or 0.4)
	P.cvx = vlat * 0.4 + (other and (sgn(P.x - other.x) ~= 0 and sgn(P.x - other.x) or 1) * rnd(3, 6) * (1 + E) or -sgn(vlat) * 3)
	P.spin = (math.random() < 0.5 and -1 or 1) * rnd(2.5, 4.5) * (0.6 + E * 0.9)
	P.vy = E > 0.25 and 2.5 + E * 9 or 0; P.air = false
	P.rollV = E > 0.25 and (math.random() < 0.5 and -1 or 1) * (2.5 + E * 5) or 0
	P.pitchV = E > 0.45 and rnd(-1, 1) * E * 5 or 0
	P.cRoll, P.cPitch = 0, 0
	if other then
		other.crashed = true; other.spinV = -P.spin * 0.5; other.change = nil
		other.v = math.max(0, other.v + (P.fwd - other.v) * (0.35 + E * 0.25))
	end
	shake = 0.6 + E; play("crash", 0.6 + 0.4 * E, 0.8)
	local e = Instance.new("Explosion"); e.BlastPressure = 0; e.BlastRadius = 1; e.DestroyJointRadiusPercent = 0
	e.Position = playerCar.root.Position + Vector3.new(0, 3, 0); e.Visible = E > 0.4; e.Parent = Workspace
end
local function takeDamage(amount, c, what)
	P.damage = math.min(what == "scrape" and SET.swipeSafe and 0.99 or 1, P.damage + amount)
	MD.feedFlow(-0.08)
	if P.damage >= 1 then
		crash(("Your car couldn't take another %s. %s finished it off at %d mph."):format(what, cap(KIND[c.kind]), math.floor(P.fwd * 2.23694 + 0.5)), c)
		return
	end
	popup(what == "scrape" and "SIDE-SWIPE" or "BUMPER TAP", ("Damage %d%%"):format(math.floor(P.damage * 100 + 0.5)))
end
function sideSwipe(c, dx, ox, toward, mult)
	local s = sgn(dx); if s == 0 then s = 1 end
	P.x += s * (ox + 0.02)
	if sgn(P.h) == -s then P.h *= -0.3 end
	P.h = clamp(P.h + s * 0.025, -0.7, 0.7); P.yaw = 0; P.si *= 0.3
	P.v *= 0.99; c.counted = true
	if c.change and sgn(c.change.to - c.change.from) == -s then
		c.change = { from = c.change.to, to = c.change.from, t = 1 - clamp(c.change.t, 0, 1) }
	end
	shake = math.max(shake, 0.35)
	if (c.scrapeCD or 0) > P.clock then return end
	c.scrapeCD = P.clock + 0.5
	play("crash", 0.25, 1.6)
	takeDamage(curDiff().dmg * (mult or 1) * (0.7 + math.min(1, toward / 5) * 0.8), c, "scrape")
end
local function bumperTap(c, oz, behind)
	if behind then c.f -= oz + 0.05; c.v = math.min(c.v, P.fwd - 1.5); P.v += 0.8
	else c.f += oz + 0.05; P.v = math.max(0, math.min(P.v, c.v - 1.5)); c.v += 1 end
	c.counted = true; shake = math.max(shake, 0.45)
	if (c.scrapeCD or 0) > P.clock then return end
	c.scrapeCD = P.clock + 0.5
	play("crash", 0.35, 1.1)
	takeDamage(curDiff().dmg * 0.6, c, "bump")
end
function nearMiss(c, clearance, headOn)
	if c.counted then return end
	c.counted = true
	P.combo = P.comboT > 0 and P.combo + 1 or 1; P.comboT = 4.5
	P.bestCombo = math.max(P.bestCombo, P.combo); P.near += 1
	local mph = P.fwd * 2.23694
	local side = sgn(c.x - P.x); if side == 0 then side = 1 end
	local thread = P.clock - P.lastNearT < 0.7 and side ~= P.lastNearSide
	P.lastNearT = P.clock; P.lastNearSide = side
	local pts = math.floor((150 + (1 - clearance) * 350) * paceMult(mph) * math.min(P.combo, 8) * (headOn and 2 or 1) * (thread and 2 or 1) * MD.flowMult() + 0.5)
	P.score += pts
	MD.feedFlow((clearance < 0.35 and 0.22 or 0.15) + (thread and 0.2 or 0))
	if thread then MD.RUSH.threads += 1 end
	popup(thread and "THREADED THE GAP" or headOn and "HEAD-ON DODGE" or clearance < 0.35 and "PAINT SWAP" or "CLOSE CALL", ("%s · %d cm · +%s"):format(KIND_UP[c.kind], math.floor(clearance * 100), commas(pts)))
	comboScale.Scale = 1.15
	play("closeCall", 0.35, 0.9 + math.min(P.combo, 8) * 0.06)
end

---------------------------------------------------------------- player physics (ported from the web version's updatePlayer)
local function stepCrash(dt)
	P.crashT += dt
	local air = P.y > 0.001 or P.vy > 0
	P.h += P.spin * dt; P.spin *= math.exp(-(air and 0.3 or 1.1) * dt)
	P.cvx *= math.exp(-(air and 0.25 or 1.6) * dt)
	local upside = math.cos(P.cRoll) * math.cos(P.cPitch) < 0
	P.cvz = math.max(0, P.cvz - (air and 1.2 or upside and 6 or 9) * dt)
	P.x += P.cvx * dt
	local lw = (SET.divided and INNER_WALL or -OUTER_WALL) + 1.4
	if P.x < lw then P.x = lw; P.cvx = math.abs(P.cvx) * 0.3 end
	if P.x > OUTER_WALL - 1.4 then P.x = OUTER_WALL - 1.4; P.cvx = -math.abs(P.cvx) * 0.3 end
	if air then
		P.vy -= 9.8 * dt; P.y += P.vy * dt
		P.cRoll += P.rollV * dt; P.cPitch += P.pitchV * dt
		if P.y <= 0 then
			P.y = 0
			local impact = -P.vy
			if impact > 2.2 then
				P.vy = impact * 0.42; P.rollV *= 0.72; P.pitchV *= 0.6; P.cvz *= 0.88
				shake = math.max(shake, 0.3 + impact * 0.06); play("land", math.min(1, 0.25 + impact * 0.08), 0.8)
			else P.vy = 0 end
		end
	else
		P.cRoll = damp(P.cRoll, math.floor(P.cRoll / math.pi + 0.5) * math.pi, 6, dt)
		P.cPitch = damp(P.cPitch, math.floor(P.cPitch / math.pi + 0.5) * math.pi, 6, dt)
	end
	P.fwd = P.cvz; P.v = P.cvz; P.f += P.cvz * dt
	P.throttle = 0
end
local function stepPlayer(dt)
	if P.crashed then stepCrash(dt); return end
	local car = CUR
	local left, right = down(Enum.KeyCode.A, Enum.KeyCode.Left), down(Enum.KeyCode.D, Enum.KeyCode.Right)
	local gas, brake = down(Enum.KeyCode.W, Enum.KeyCode.Up), down(Enum.KeyCode.S, Enum.KeyCode.Down, Enum.KeyCode.Space)
	local airborne = P.air
	local steerIn = (right and 1 or 0) - (left and 1 or 0)
	local brk = brake and 1 or 0
	-- the virtual wheel turns at a limited rate, input is eased, and yaw follows with a little lag
	local sens = SET.steer
	local reversing = steerIn ~= 0 and P.si ~= 0 and sgn(steerIn) ~= sgn(P.si)
	local wheelRate = (steerIn == 0 and 3.2 + sens * 1.6 or reversing and 4.5 + sens * 2.5 or 1.6 + sens * 1.9) * car.steer
	P.si += clamp(steerIn - P.si, -wheelRate * dt, wheelRate * dt)
	local siEased = P.si * (0.35 + 0.65 * math.abs(P.si))
	P.throttle = damp(P.throttle, gas and 1 or 0, 8, dt)
	local v = math.max(P.v, 0.1)
	local wet = SET.weather == "rain" and 0.85 or 1
	local maxYaw = math.min(0.9 + sens * 0.45, ((6.2 + sens * 2.8) * car.grip * wet) / math.max(v, 4))
	-- with auto-straighten off, letting go holds the current heading
	local centring = SET.autoCenter and clamp(-P.h * (1.8 + sens * 1.0), -maxYaw, maxYaw) or 0
	local yawTarget = math.abs(P.si) > 0.02 and siEased * maxYaw or centring
	if airborne then yawTarget *= 0.35 end
	P.yaw = damp(P.yaw, yawTarget, (4.5 + sens * 3) * car.steer, dt)
	local lat, grip = P.yaw * v, (8.4 + sens * 1.6) * car.grip * wet
	if math.abs(lat) > grip then P.yaw *= grip / math.abs(lat) end
	P.h = clamp(P.h + P.yaw * dt, -0.7, 0.7)
	-- in a bend the road turns under you: hold your line by steering into it
	if TERRAIN.bend ~= 0 then
		local kx = (hillX(P.f + 1) - 2 * hillX(P.f) + hillX(P.f - 1)) * TERRAIN.bend
		P.h = clamp(P.h - P.v * kx * 0.4 * dt, -0.7, 0.7)
	end
	-- longitudinal
	local m, power = 1520, car.power
	local drive = airborne and 0 or P.throttle * math.min(m * (5.4 + power / 1.2e6), power / math.max(v, 3))
	local drag, rollR, engBrake = car.drag * v * v, m * 9.81 * 0.013, (1 - P.throttle) * m * 0.55
	local brakeF = (airborne and 0 or brk) * m * (12.5 + math.min(v, 80) * 0.035) * car.brake * wet
	-- hills: gravity along the slope
	local slope = TERRAIN.amp ~= 0 and (hillH(P.f + 0.5) - hillH(P.f - 0.5)) * TERRAIN.amp or 0
	local a = (drive - drag - rollR - engBrake - brakeF) / m - math.abs(lat) * 0.025 - 9.81 * slope
	P.boosting = SET.mode == "stunt" and P.nitro > 0 and down(Enum.KeyCode.LeftShift, Enum.KeyCode.RightShift, Enum.KeyCode.N)
	if P.boosting then a += 15; P.nitro = math.max(0, P.nitro - dt * 0.2) end
	P.v = clamp(P.v + a * dt, 0, P.boosting and car.top * 1.22 or math.max(car.top, P.v - 10 * dt))
	-- gears & rpm
	local rpm = (P.v / TIRE_C) * 60 * GEARS[P.gear] * car.final
	if rpm > 7400 and P.gear < #GEARS then P.gear += 1 elseif rpm < 3200 and P.gear > 1 then P.gear -= 1 end
	rpm = math.max(900 + P.throttle * 600, (P.v / TIRE_C) * 60 * GEARS[P.gear] * car.final)
	P.rpm = damp(P.rpm, math.min(rpm, 7900), 12, dt)
	-- integrate
	P.x += math.sin(P.h) * P.v * dt
	P.fwd = math.cos(P.h) * P.v
	P.f += P.fwd * dt
	P.top = math.max(P.top, P.fwd)
	-- guardrails: glance off at shallow angles, wreck if you drive straight into one
	local hw = (playerCar.W / 2) * math.cos(P.h) + (playerCar.L / 2) * math.abs(math.sin(P.h))
	local function hitWall(side)
		local vlat = math.sin(P.h) * P.v
		if math.abs(vlat) > 10 then P.wall = true; crash(("You hit the %s at %d mph."):format(side < 0 and SET.divided and "median barrier" or "guardrail", math.floor(P.fwd * 2.23694 + 0.5))); return end
		P.h = -P.h * 0.25; P.yaw = 0; P.v *= 0.6 ^ (dt * 6); shake = math.max(shake, 0.25)
		play("crash", 0.12, 1.8)
	end
	local lw = SET.divided and INNER_WALL or -OUTER_WALL
	if P.x - hw < lw then P.x = lw + hw; hitWall(-1)
	elseif P.x + hw > OUTER_WALL then P.x = OUTER_WALL - hw; hitWall(1) end
end

local function collisions()
	if P.crashed then return end
	local hw = (playerCar.W / 2) * math.cos(P.h) + (playerCar.L / 2) * math.abs(math.sin(P.h))
	local hl = (playerCar.L / 2) * math.cos(P.h) + (playerCar.W / 2) * math.abs(math.sin(P.h))
	local vlat = math.sin(P.h) * P.v
	for _, c in ipairs(traffic) do
		local dz = c.f - P.f -- + ahead
		local cwid = c.W / 2 + math.abs(math.sin(c.yawv or 0)) * c.L / 2
		local dx = P.x - c.x
		local ox, oz = hw + cwid - 0.05 - math.abs(dx), hl + c.L / 2 - 0.05 - math.abs(dz)
		local clearH = c.kind == "semi" and 4.1 or (c.kind == "van" and 2.3 or 1.55)
		if ox > 0 and oz > 0 and P.y > clearH then
			if not (P.clock - (c.jumpT or -9) < 3) then c.jumpT = P.clock; P.cleared += 1 end
			continue
		end
		if ox > 0 and oz > 0 and not c.crashed then
			local toward = math.max(0, -sgn(dx) * vlat)
			if ox < oz and ox < 0.55 and toward < 7 then sideSwipe(c, dx, ox, toward); continue end
			if ox >= oz and oz < 0.9 and math.abs(P.fwd - c.v) < 4.5 then bumperTap(c, oz, dz < 0); continue end
		end
		if ox > 0 and oz > 0 then
			local rear = dz < 0 and down(Enum.KeyCode.S, Enum.KeyCode.Down, Enum.KeyCode.Space)
			local mph = math.floor(P.fwd * 2.23694 + 0.5)
			crash(rear and ("%s rear-ended you at %d mph."):format(cap(KIND[c.kind]), mph) or ("You clipped %s at %d mph."):format(KIND[c.kind], mph), c)
			return
		end
		-- close call: the player's front passes the car's rear (or the reverse)
		local side = dz < 0 and -1 or 1
		if side ~= c.side and math.abs(dz) < c.L / 2 + 2.6 then
			local hw2 = playerCar.W / 2 + (playerCar.L / 2) * math.abs(math.sin(P.h))
			local clearance = math.abs(c.x - P.x) - hw2 - c.W / 2
			if clearance < 1.0 and clearance > -0.1 and P.fwd > 18 then nearMiss(c, clearance) end
		end
		c.side = side
	end
end

---------------------------------------------------------------- traffic
local function stepTraffic(dt)
	updateDiff()
	for i = #traffic, 1, -1 do
		local c = traffic[i]
		local target = c.crashed and 0 or c.desired
		if not c.crashed then
			for _, o in ipairs(traffic) do
				if o ~= c and math.abs(o.x - c.x) < 2.4 and o.f > c.f then
					local gap = o.f - c.f - (o.L + c.L) / 2
					if gap < 8 + c.v * 1.1 then target = math.min(target, o.v - (8 + c.v - gap) * 0.05) end
				end
			end
			-- don't drive into the player's car either
			if math.abs(P.x - c.x) < 2.2 and P.f > c.f and not P.crashed then
				local gap = P.f - c.f - (playerCar.L + c.L) / 2
				if gap < 6 + c.v * 0.6 then target = math.min(target, P.fwd - 1) end
			end
			if c.yb then target = math.min(target, math.max(0, P.v * 0.72)) end
		end
		c.v = math.max(0, c.v + clamp(target - c.v, -9 * dt, 3 * dt))
		c.f += c.v * dt
		if c.crashed then c.yawv = (c.yawv or 0) + (c.spinV or 0) * dt; c.spinV = (c.spinV or 0) * math.exp(-1.5 * dt) end
		-- random lane changes, more often on harder settings and farther along
		if not c.crashed and not c.change then
			c.timer -= dt
			if c.timer <= 0 then
				c.timer = rnd(3, 8)
				if math.random() < DIFF.lane then
					local to = c.lane + (math.random() < 0.5 and -1 or 1)
					if to >= 1 and to <= 4 and not (c.kind == "semi" and to == 1) and laneFree(to, c.f, c, 10 + c.v * 0.5) then
						c.change = { from = c.lane, to = to, t = -0.4 }
					end
				end
			end
		end
		if c.change then
			c.change.t += dt / (c.change.fast and 1.8 or 2.8)
			local t = clamp(c.change.t, 0, 1)
			local e = t * t * (3 - 2 * t)
			local nx = laneX(c.change.from) + (laneX(c.change.to) - laneX(c.change.from)) * e
			if dt > 0 then c.yawv = c.change.t > 0 and -(nx - c.x) / math.max(dt, 1e-3) / math.max(c.v, 5) or 0 end
			c.x = nx
			if c.change.t >= 1 then c.lane = c.change.to; c.x = laneX(c.lane); c.change = nil; c.yawv = 0 end
		end
		if c.f < P.f - 160 or c.f > P.f + 520 then c.model:Destroy(); table.remove(traffic, i) end
	end
	local tries = 0
	while #traffic < DIFF.traffic and tries < 6 do
		tries += 1
		if P.fwd < 24 then -- crawling: let traffic come up from behind
			local lane = math.random(1, 4)
			if math.abs(laneX(lane) - P.x) > 2.5 then spawnTraffic(P.f - rnd(70, 110), lane) end
		else spawnTraffic(P.f + (SET.mode == "rush" and rnd(150, 470) or rnd(260, 440))) end
	end
	while #traffic > DIFF.traffic + 4 do -- slider turned down: drop the farthest car ahead
		local far, fi = -1e9, nil
		for i, c in ipairs(traffic) do if c.f > far and c.f > P.f + 150 then far = c.f; fi = i end end
		if not fi then break end
		traffic[fi].model:Destroy(); table.remove(traffic, fi)
	end
end

local function yieldToIndicator(dt)
	if ind == 0 or state ~= "play" or P.crashed then return end
	indT += dt
	local pl, bd = 2, 99
	for i = 1, 4 do local d = math.abs(P.x - laneX(i)); if d < bd then bd = d; pl = i end end
	if bd > 2.6 then return end
	local tl = pl + ind
	if tl < 1 or tl > 4 then return end
	for _, c in ipairs(traffic) do
		if c.lane == tl and not c.change and not c.crashed and c.f - P.f > -30 and c.f - P.f < 60 then
			c.yr = c.yr or (indT + rnd(0.4, 1.3))
			if indT >= c.yr then
				local fl = tl + ind
				if fl >= 1 and fl <= 4 and laneFree(fl, c.f, c, 12 + c.v * 0.5) then c.change = { from = c.lane, to = fl, t = -0.25, fast = true }; c.yb = false
				else c.yb = c.f - P.f < 8 end
			end
		elseif c.yb and c.lane ~= tl then c.yb = false end
	end
end

			local a = math.sin(P.crashT * 0.7) * (0.4 + E * 0.9)
			want = { x = P.x + math.sin(a) * (7 + E * 7), y = 2.3 + E * 3.5 + P.y * 0.5, f = P.f - math.cos(a) * (7 + E * 7) }
			look = w2v(P.x, P.y + 0.5, P.f + 1)
			if not CR then CR = want end
			local k = 1 - math.exp(-2.2 * dt)
			CR = { x = CR.x + (want.x - CR.x) * k, y = CR.y + (want.y - CR.y) * k, f = CR.f + (want.f - CR.f) * k }
		else
			local back, height = 6.2 + math.min(sp, 90) * 0.022, 2.05 + math.min(sp, 90) * 0.004
			local head = P.h * 0.55
			want = { x = P.x - math.sin(head) * back, y = height + P.y * 0.8, f = P.f - math.cos(head) * back }
			if not CR then CR = want end
			-- distance behind is damped relative to the car so the camera never falls miles back
			local rel = damp(CR.f - P.f, want.f - P.f, 4, dt)
			CR = { x = damp(CR.x, want.x, 7, dt), y = damp(CR.y, want.y, 5, dt), f = P.f + rel }
			look = w2v(P.x + math.sin(head) * 12, 0.95 + P.y * 0.9, P.f + 12)
		end
		local cf = CFrame.lookAt(w2v(CR.x, CR.y, CR.f), look)
		local s = (clamp(sp / 140, 0, 1) ^ 2) * 0.024 + shake * 0.25
		local t = os.clock()
		cf = cf * CFrame.new((math.sin(t * 37) + math.sin(t * 23.3)) * s * 0.5 * S, (math.sin(t * 41.7) + math.sin(t * 29.1)) * s * 0.4 * S, 0)
		camera.CFrame = cf
		setCarVisible(car, true); if cockpit.folder then cockpit.folder.Parent = nil end
	else
		CR = nil
		local eye = car.root.CFrame * CFrame.new(-0.4 * S, 1.2 * S, -0.1 * S)
		local s = shake * 0.12
		camera.CFrame = eye * CFrame.new((math.random() - 0.5) * s * S, (math.random() - 0.5) * s * S, 0)
		setCarVisible(car, false); layoutCockpit(camera.CFrame); cockpit.folder.Parent = world
	end
	local fov = (cm == 0 and 56 + clamp(sp, 0, 145) * 0.13 or 66 + clamp(sp, 0, 145) * 0.07) + (P.boosting and 7 or 0)
	camera.FieldOfView = damp(camera.FieldOfView, fov, 3, dt)
	shake = math.max(0, shake - dt * 1.6)
end

local PUR = { heat = 1, phase = "off", coolT = 0, bust = 0, evadeT = 0, escapes = 0, downs = 0, nearest = 1e9, nearestZ = 0, nearestUnit = 0, copMax = 70 }
MD.PUR = PUR

local function spawnWave()
	PUR.copMax = (56 + PUR.heat * 13) * curDiff().cop
	local n = math.min(#cops, 1 + math.ceil(PUR.heat * 0.85))
	local offsets = { 0, -2.6, 2.6, 0, -2.6, 2.6 }
	for i = 1, n do
		local c = cops[i]
		local ahead = PUR.heat >= 3 and i == n -- from heat 3 one unit cuts in from the front
		c.active = true; c.crashed = false; c.vx = 0; c.yaw = 0; c.spinV = 0; c.hitCD = 0; c.offset = offsets[i]; c.unit = 10 + PUR.heat * 10 + i
		c.x = laneX(i % 4 + 1); c.tx = c.x
		c.f = ahead and P.f + rnd(520, 650) or P.f - 150 - (i - 1) * 40 - rnd(0, 25)
		c.v = ahead and math.max(20, P.fwd * 0.55) or P.fwd + 6
		c.car.model.Parent = world
	end
	PUR.phase = "chase"; PUR.evadeT = 0
	popup(("HEAT %d"):format(PUR.heat), ("%d units in pursuit · top speed %d mph"):format(n, math.floor(PUR.copMax * 2.23694 + 0.5)))

local function lanesClear(x, f0, f1)
	for _, t in ipairs(traffic) do if math.abs(t.x - x) < 2.4 and t.f + t.L / 2 > f0 and t.f - t.L / 2 < f1 then return false end end
	return true
end
local function wreckCop(c, other)
	c.crashed = true; c.spinV = (math.random() < 0.5 and -1 or 1) * rnd(2, 4)
	if other then other.crashed = true; other.spinV = -c.spinV * 0.4; other.change = nil end
	if state == "play" and not P.crashed then
		PUR.downs += 1
		local pts = 1500 * PUR.heat; P.score += pts
		popup("UNIT DOWN", ("Unit %d hit %s · +%s"):format(c.unit, other and KIND_A[other.kind] or "oncoming traffic", commas(pts)))
		play("crash", 0.5, 0.7)
	end
end
function MD.updateCops(dt)
	for _, c in ipairs(cops) do
		if not c.active then continue end
		c.hitCD -= dt
		if c.crashed then
			c.v = math.max(0, c.v - 12 * dt); c.yaw += c.spinV * dt; c.spinV *= math.exp(-1.5 * dt)
			c.f += c.v * dt
			if c.f < P.f - 400 or c.f > P.f + 1200 then c.active = false; c.car.model.Parent = nil end
		else
			local dz = P.f - c.f -- + = cop behind the player
			local max = PUR.copMax * (dz > 220 and 1.15 or 1)
			local target = dz > 0 and math.min(max, P.fwd + clamp(dz * 0.16, 5, 32)) or math.max(8, math.min(max, P.fwd - clamp(-dz * 0.22, -3, 12)))
			local tx = math.abs(dz) < 90 and P.x + c.offset or c.tx
			-- dodge traffic in the cruiser's path
			local blocker, gap = nil, 1e9
			for _, t in ipairs(traffic) do
				if t.f > c.f and math.abs(t.x - c.x) < (t.W + 1.9) / 2 + 0.35 then
					local g = (t.f - t.L / 2) - (c.f + 2.4)
					if g < gap then gap = g; blocker = t end
				end
			end
			if blocker and gap < 14 + c.v * 0.7 then
				local best, bd = nil, 1e9
				for l = 1, 4 do
					if math.abs(laneX(l) - blocker.x) > 2.4 and lanesClear(laneX(l), c.f - 8, blocker.f + 35) then
						local d = math.abs(laneX(l) - c.x); if d < bd then bd = d; best = l end
					end
				end
				if best then c.tx = laneX(best); tx = c.tx else target = math.min(target, blocker.v - 1) end
			end
			c.v += clamp(target - c.v, -10 * dt, 7 * dt)
			c.vx = damp(c.vx, clamp((tx - c.x) * 2.4, -7, 7), 4, dt)
			c.x = clamp(c.x + c.vx * dt, SET.divided and 1.5 or -18.4, 18.4)
			c.yaw = -c.vx / math.max(c.v, 8)
			c.f += c.v * dt
			for _, t in ipairs(traffic) do
				if math.abs(t.x - c.x) < (t.W + c.car.W) / 2 - 0.15 and math.abs(t.f - c.f) < (t.L + c.car.L) / 2 - 0.15 then wreckCop(c, t); break end
			end
			if not SET.divided and not c.crashed then
				for _, o in ipairs(oncoming) do
					if math.abs(o.x - c.x) < (o.W + c.car.W) / 2 - 0.15 and math.abs(o.f - c.f) < (o.L + c.car.L) / 2 - 0.15 then wreckCop(c, nil); o.crashed = true; o.spinV = rnd(-3, 3); break end
				end
			end
			for _, o in ipairs(cops) do
				if o ~= c and o.active and not o.crashed and math.abs(o.x - c.x) < 2.0 and math.abs(o.f - c.f) < 5 then c.x += (sgn(c.x - o.x) ~= 0 and sgn(c.x - o.x) or 1) * 2.5 * dt end
			end
		end
		poseCar(c.car, c.x, c.f, c.yaw, 0, 0)
	end
	-- light bars: red and blue in turns, with a fast flicker
	local now = os.clock() * 1000
	local f, flick = math.floor(now / 95) % 6, now % 95 < 62
	local redOn, blueOn = f < 3 and flick, f >= 3 and flick
	for _, c in ipairs(cops) do
		if c.active then
			local on = not c.crashed or (now % 400 < 200)
			c.car.red.Color = (redOn and on) and Color3.fromRGB(255, 30, 30) or Color3.fromRGB(70, 0, 0)
			c.car.blue.Color = (blueOn and on) and Color3.fromRGB(50, 90, 255) or Color3.fromRGB(0, 10, 70)
			c.car.redL.Brightness = (redOn and on) and 4 or 0; c.car.blueL.Brightness = (blueOn and on) and 4 or 0
		end
	end
end
function MD.checkCopHits()
	if P.crashed then return end
	local hw = (playerCar.W / 2) * math.cos(P.h) + (playerCar.L / 2) * math.abs(math.sin(P.h))
	local hl = (playerCar.L / 2) * math.cos(P.h) + (playerCar.W / 2) * math.abs(math.sin(P.h))
	for _, c in ipairs(cops) do
		if c.active and not c.crashed and c.hitCD <= 0 and math.abs(c.x - P.x) < hw + c.car.W / 2 - 0.05 and math.abs(c.f - P.f) < hl + c.car.L / 2 - 0.05 then
			c.hitCD = 0.8
			PUR.bust = math.min(1, PUR.bust + 0.18)
			local side = sgn(P.x - c.x); if side == 0 then side = 1 end
			if math.abs(P.x - c.x) > 1.3 then P.h = clamp(P.h + side * 0.05, -0.7, 0.7); c.vx -= side * 3 end
			P.v *= 0.93; c.v = math.max(0, math.min(c.v, P.fwd) - 3); c.f += c.f <= P.f and -0.6 or 0.6
			shake = math.max(shake, 0.6); play("crash", 0.45, 0.9)
			if PUR.bust >= 1 then MD.busted(c) end
			return
		end
	end
end
function MD.busted(c)
	if state ~= "play" then return end
	P.busted = true; P.msg = ("Unit %d boxed you in at %d mph."):format(c and c.unit or PUR.nearestUnit, math.floor(P.fwd * 2.23694 + 0.5))
	MD.finish()
end
local function escaped()
	local bonus = 2500 * PUR.heat
	P.score += bonus; PUR.escapes += 1
	popup("ESCAPED", "+" .. commas(bonus) .. " · " .. (PUR.heat < MAX_HEAT and "heat rising" or "max heat"))
	play("closeCall", 0.5, 1.2)
	clearCops()
	PUR.heat = math.min(MAX_HEAT, PUR.heat + 1)
	PUR.phase = "cool"; PUR.coolT = 6; PUR.evadeT = 0; PUR.bust = 0
end
function MD.updatePursuitRules(dt)
	local nearest, nearCount, nz, unit, chasers, lost = 1e9, 0, 0, 0, 0, true
	for _, c in ipairs(cops) do
		if c.active and not c.crashed then
			chasers += 1
			local dz = P.f - c.f
			local d = math.sqrt(dz * dz + (c.x - P.x) ^ 2)
			if d < nearest then nearest = d; nz = dz; unit = c.unit end
			if d < 14 then nearCount += 1 end
			if dz < 320 and dz > -500 then lost = false end
		end
	end
	PUR.nearest = nearest; PUR.nearestZ = nz; PUR.nearestUnit = unit
	if PUR.phase == "cool" then PUR.coolT -= dt; if PUR.coolT <= 0 then spawnWave() end return end
	if PUR.phase ~= "chase" then return end
	if nearest < 14 and P.fwd < 22 then PUR.bust += 0.4 * dt
	elseif nearCount >= 2 and P.fwd < 40 then PUR.bust += 0.2 * dt
	else PUR.bust -= 0.1 * dt end
	PUR.bust = clamp(PUR.bust, 0, 1)
	if PUR.bust >= 1 then MD.busted(nil); return end
	if chasers == 0 or lost then PUR.evadeT += dt else PUR.evadeT = math.max(0, PUR.evadeT - dt * 2) end
	if PUR.evadeT >= EVADE_TIME then escaped() end
end

-- ---------- Rush hour ----------
local RUSH = { flow = 1, cuts = 0, threads = 0, lane = -1, warned = false }
MD.RUSH = RUSH
function MD.resetRush() RUSH.flow = 1; RUSH.cuts = 0; RUSH.threads = 0; RUSH.lane = -1; RUSH.warned = false end
function MD.flowMult() return SET.mode ~= "rush" and 1 or RUSH.flow >= 0.8 and 3 or RUSH.flow >= 0.45 and 2 or 1 end
function MD.feedFlow(a) if SET.mode == "rush" then RUSH.flow = clamp(RUSH.flow + a, 0, 1) end end
local function cutUp(c, dz)
	RUSH.cuts += 1
	P.comboT = math.max(P.comboT, 4.5) -- a cut keeps the close-call chain alive
	local gap = math.max(0, math.abs(dz) - c.L / 2 - 2.31)
	local pts = math.floor((250 + (1 - math.min(gap, 11) / 11) * 450) * paceMult(P.fwd * 2.23694) * MD.flowMult() * math.max(1, math.min(P.combo, 8)) + 0.5)
	P.score += pts; MD.feedFlow(gap < 3 and 0.16 or 0.1)
	popup(gap < 3 and "TIGHT CUT" or "CUT UP", ("%s %s · %.1f m · +%s"):format(dz > 0 and "in front of" or "behind", KIND_A[c.kind], gap, commas(pts)))
	play("closeCall", 0.35, 1.1)
end
function MD.updateRush(dt)
	-- a cut-up is moving into a lane that has a car within ~11 m of your bumpers
	local li, bd = -1, 1.0
	for i = 1, 4 do local d = math.abs(P.x - laneX(i)); if d < bd then bd = d; li = i end end
	if li >= 0 and li ~= RUSH.lane then
		if RUSH.lane >= 0 and P.fwd > 18 then
			local tight, tz = nil, 1e9
			for _, c in ipairs(traffic) do
				local dz = c.f - P.f
				if not c.crashed and math.abs(c.x - laneX(li)) <= 1.9 and math.abs(dz) <= c.L / 2 + 13 and math.abs(dz) < math.abs(tz) then tight = c; tz = dz end
			end
			if tight then cutUp(tight, tz) end
		end
		RUSH.lane = li
	end
	local mph = P.fwd * 2.23694
	if P.clock > 4 then RUSH.flow -= (0.05 + math.min(0.035, (P.f / MILE) * 0.006)) * (mph < 50 and 2.2 or 1) * curDiff().drain * dt end
	if RUSH.flow < 0.25 and not RUSH.warned then RUSH.warned = true; popup("KEEP WEAVING", "Your flow is running out")
	elseif RUSH.flow > 0.4 then RUSH.warned = false end
	if RUSH.flow < 0 then RUSH.flow = 0 end -- an empty meter only drops the multiplier to ×1
end

---------------------------------------------------------------- game flow
applyTod = function()
	todNow = SET.tod; rainOn = SET.weather == "rain"
	applyWeather(); setWet(rainOn)
	local T = ({ flat = { 0, 0 }, hills = { 13, 0 }, curves = { 13, 45 } })[SET.terrain] or { 0, 0 }
	TERRAIN.amp, TERRAIN.bend = T[1], T[2]
	setDivided(SET.divided)
end
showCar = function()
	if playerCar then playerCar.model:Destroy() end
	CUR = carDef(SET.car)
	playerCar = buildCar(CUR.id, PAINTS[SET.paint].c)
end
local function resetRun()
	clearTraffic(); clearStunt()
	P = newP(); P.v = 22; P.fwd = 22
	updateDiff()
	for _ = 1, 90 do
		if #traffic < DIFF.traffic then
			local lane, f = math.random(1, 4), SET.mode == "rush" and rnd(30, 460) or rnd(45, 430)
			if not (math.abs(laneX(lane) - P.x) < 2.5 and f < 80) then spawnTraffic(f, lane) end -- keep your own lane clear at the start
		end
	end
	MD.resetOncoming(); MD.resetPursuit(); MD.resetRush()
	nextRampF = 260; nextPickF = 120
	hist = {}; histT = 0; rewinding = nil; ind = 0; camPos = nil; camMode = camMode
	for _, p in ipairs(popList) do p.g:Destroy() end; table.clear(popList)
end
local TUT -- tutorial state, set further down
startGame = function(tutorial)
	if tutorial == true or not SET.tutorialDone then TUT.start() else TUT.stop() end
	if not unlocked(CUR) then SET.car = "vanta"; CUR = carDef("vanta") end
	if not playerCar or playerCar.kind ~= CUR.id then showCar() end
	resetRun()
	state = "play"; menu.Visible = false; over.Visible = false; pauseO.Visible = false; hud.Visible = true
	rewFx.Visible = false
	popup(({ endless = "ENDLESS", stunt = "STUNT RUN", pursuit = "PURSUIT", rush = "RUSH HOUR" })[SET.mode], "Faster driving multiplies points")
	hintRun += 1; local my = hintRun; hint.Visible = true
	task.delay(12, function() if hintRun == my then hint.Visible = false end end)
end
toMenu = function()
	state = "menu"; clearTraffic(); clearStunt(); MD.clearOncoming(); MD.resetPursuit(); P = newP(); showCar(); TUT.stop()
	menu.Visible = true; over.Visible = false; pauseO.Visible = false; hud.Visible = false; rewFx.Visible = false
	rewinding = nil; camPos = nil; refreshMenu()
end
local function bank()
	local sc = math.floor(P.score)
	local m = SET.mode
	local newBest = sc > (PROG.best[m] or 0)
	if sc > 0 then
		PROG.career += sc
		if newBest then PROG.best[m] = sc end
		local ev = RS:FindFirstChild("IWSubmit"); if ev then ev:FireServer(m, sc) end
	end
	return sc, newBest
end
function finishGame()
	state = "over"
	local before = {}
	for _, c in ipairs(CARS) do before[c.id] = unlocked(c) end
	local sc, newBest = bank()
	overTitle.Text = P.busted and "BUSTED" or P.wall and "INTO THE WALL" or P.crashE > 0.6 and "TOTALLED" or "WRECKED"
	local fresh = {}
	for _, c in ipairs(CARS) do if unlocked(c) and not before[c.id] then fresh[#fresh + 1] = c.name end end
	overMsg.Text = P.msg .. (#fresh > 0 and ("  Unlocked: " .. table.concat(fresh, ", ") .. "!") or "")
	statVals["Score"].Text = commas(sc)
	statVals["Best"].Text = commas(PROG.best[SET.mode] or 0) .. (newBest and " ★" or "")
	statVals["Distance"].Text = ("%.1f mi"):format(P.f / MILE)
	statVals["Top speed"].Text = ("%d mph"):format(math.floor(P.top * 2.23694 + 0.5))
	statVals["Close calls"].Text = tostring(P.near)
	statVals["Best combo"].Text = "×" .. tostring(P.bestCombo)
	if SET.mode == "stunt" then statVals["Best combo"].Text ..= ("  ·  %d jumps"):format(P.jumps) end
	if SET.mode == "rush" then overMsg.Text ..= (" %d cut-up%s, %d gap%s threaded."):format(MD.RUSH.cuts, MD.RUSH.cuts == 1 and "" or "s", MD.RUSH.threads, MD.RUSH.threads == 1 and "" or "s") end
	if SET.mode == "pursuit" then overMsg.Text ..= (" Escaped %d time%s at heat %d, took down %d unit%s."):format(MD.PUR.escapes, MD.PUR.escapes == 1 and "" or "s", MD.PUR.heat, MD.PUR.downs, MD.PUR.downs == 1 and "" or "s") end
	over.Visible = true
end
MD.finish = function()
	if P.busted then P.crashed = true; P.cvz = P.fwd * 0.4; P.crashE = 0 end
	finishGame()
end
restartRun = function()
	if state ~= "play" and state ~= "paused" and state ~= "over" then return end
	-- keep what this run earned before throwing it away
	local sc = 0
	if state ~= "over" then sc = bank() end
	startGame()
	if sc > 0 then popup("RUN BANKED", "+" .. commas(sc) .. " career pts kept") end
end
togglePause = function()
	if state == "play" then state = "paused"; pauseO.Visible = true
	elseif state == "paused" then state = "play"; pauseO.Visible = false; setO.Visible = false end
end
local function startRewind()
	if #hist < 4 or rewinding then return end
	TUT.event("rew")
	if state == "over" then over.Visible = false; state = "play" end
	if SET.rewindFx then
		rewinding = { i = #hist, acc = 0, n = #hist }
		rewFx.Visible = true; play("tick", 0.4, 0.6)
	else
		restore(hist[1]); for k = #hist, 2, -1 do hist[k] = nil end
		P.crashed = false; popup("REWIND", "Back 7 seconds"); doFlash(0.5)
	end
end

RunService.RenderStepped:Connect(function(dtRaw)
	local dt = math.min(dtRaw, 0.05)
	camera.CameraType = Enum.CameraType.Scriptable
	VOL.sfx = SET.sfx; VOL.muted = SET.muted
	if not playerCar then showCar() end
	if state == "menu" then
		poseCar(playerCar, P.x, 0, 0, 0, 0)
		updateRoad(0); updateCamera(dt)
		rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 26, -12)); rain.Rate = rainOn and 1400 or 0
		blurFx.Size = 0
		return
	end
	if state == "paused" then return end

	if rewinding then
		-- scrub backwards through the snapshots like a tape, easing in and out
		rewinding.acc += dt / (0.5 + 0.08 * rewinding.n / 10)
		local x = math.min(1, rewinding.acc)
		local e = x < 0.5 and 2 * x * x or 1 - (-2 * x + 2) ^ 2 / 2
		local idx = math.max(1, math.floor((1 - e) * (rewinding.n - 1) + 1.5))
		restore(hist[idx])
		rewSec.Text = ("-%.1f s"):format((rewinding.n - 1) / 10 * e)
		scan.Position = UDim2.fromScale(0, (os.clock() * 1.8) % 1.2 - 0.14)
		if x >= 1 then
			for k = #hist, 2, -1 do hist[k] = nil end
			restore(hist[1]); histT = 0
			rewinding = nil; rewFx.Visible = false; P.crashed = false; camPos = nil
			doFlash(0.5); popup("REWIND", (#hist >= 1 and "Back 7 seconds" or ""))
		end
	elseif state == "play" or state == "over" then
		P.clock += dt
		stepPlayer(dt)
		if not P.crashed then updateStunt(dt) end
		yieldToIndicator(dt); stepTraffic(dt); MD.updateOncoming(dt); collisions(); MD.checkOncoming()
		if SET.mode == "pursuit" then
			MD.updateCops(dt)
			if state == "play" and not P.crashed then MD.checkCopHits(); MD.updatePursuitRules(dt) end
		end
		if SET.mode == "rush" and state == "play" and not P.crashed then MD.updateRush(dt) end
		TUT.update(dt)
		if not P.crashed then
			local wrongWay = not SET.divided and P.x < 0 -- driving on the oncoming side pays double
			P.score += P.fwd * dt * 0.1 * paceMult(P.fwd * 2.23694) * (wrongWay and 2 or 1) * MD.flowMult()
			P.comboT = math.max(0, P.comboT - dt)
			histT += dt
			if histT >= 0.1 then
				histT -= 0.1; hist[#hist + 1] = snapshot()
				if #hist > REWIND_S * 10 + 1 then table.remove(hist, 1) end
			end
		elseif state == "play" and P.crashT > 2.2 then finishGame() end
	end
	flash.BackgroundTransparency = math.min(1, flash.BackgroundTransparency + dt * 2)

	updateRoad(P.f)
	poseCar(playerCar, P.x, P.f, -P.h, P.crashed and P.cRoll or -P.yaw * 0.06, P.y, P.crashed and P.cPitch or (P.air and clamp(P.vy * 0.03, -0.3, 0.3) or 0))
	for _, c in ipairs(traffic) do poseCar(c, c.x, c.f, c.yawv or 0, 0, 0) end
	updateIndicators(); updateCamera(dt); updateHud(dt); MD.updateHud()
	-- music and engine loops (only if audio ids are set in SOUND_IDS)
	for _, k in ipairs({ "night", "lofi", "overdrive" }) do
		local s = SOUNDS[k]
		if s then
			local on = SET.track == k and not SET.muted and SET.music > 0
			if on and not s.IsPlaying then s:Play() elseif not on and s.IsPlaying then s:Stop() end
			s.Volume = SET.music * 0.5
		end
	end
	local eng = SOUNDS.engine
	if eng then
		local on = state == "play" and not SET.muted
		if on and not eng.IsPlaying then eng:Play() elseif not on and eng.IsPlaying then eng:Stop() end
		eng.PlaybackSpeed = 0.6 + P.rpm / 8000 * 1.4; eng.Volume = SET.sfx * (0.25 + P.throttle * 0.35)
	end
	-- motion blur at speed (Roblox has no radial blur, so this is a gentle full-screen blur)
	local lvl = SET.motionBlur == "strong" and 1 or SET.motionBlur == "light" and 0.5 or 0
	local xs = clamp((P.fwd - 26) / 70, 0, 1)
	blurFx.Size = (state == "play" and not P.crashed) and lvl * 5 * xs * xs * (3 - 2 * xs) + (P.boosting and lvl * 2 or 0) or 0
	rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 26, -12)) * CFrame.Angles(-math.atan2(P.v * S, 150) * 0.6, 0, 0)
	rain.Rate = rainOn and 1400 or 0
end)
