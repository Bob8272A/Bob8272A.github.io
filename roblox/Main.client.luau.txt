	end)

local function mk(parent, size, cf, color, mat, o)
	local p = Instance.new("Part")
	p.Anchored = true; p.CanCollide = false; p.CanTouch = false; p.CanQuery = false
	p.Size = size * S; p.CFrame = cf; p.Color = color
	p.Material = mat or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth; p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		if o.tr then p.Transparency = o.tr end
		if o.refl then p.Reflectance = o.refl end
		if o.shape then p.Shape = o.shape end
		if o.shadow == false then p.CastShadow = false end
	end
	p.Parent = parent
	return p
end

local function cp(car, sx, sy, sz, ox, oy, oz, color, mat, o)
	local rot = CFrame.new()
	if o and o.rx then rot = CFrame.Angles(math.rad(o.rx), 0, 0) end
	local p = mk(car.model, Vector3.new(sx, sy, sz), car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * rot, color, mat, o)
	if color == car.paint and mat == nil then p.Reflectance = 0.12 end
	p.Anchored = false; p.Massless = true
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = p; wc.Parent = p
	return p
end

local GLASS = Color3.fromRGB(18, 28, 40)
-- sloped glass/panel welded to a car (tall end toward +Z by default; flip = tall end toward -Z)
local function cw(car, sx, sy, sz, ox, oy, oz, color, mat, flip, o)
	local w = Instance.new("WedgePart"); w.Anchored = false; w.CanCollide = false; w.CanTouch = false; w.CanQuery = false; w.Massless = true
	w.Size = Vector3.new(sx, sy, sz) * S; w.Color = color; w.Material = mat or Enum.Material.SmoothPlastic
	if o and o.refl then w.Reflectance = o.refl elseif color == car.paint and mat == nil then w.Reflectance = 0.12 end
	w.CFrame = car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * (flip and CFrame.Angles(0, math.pi, 0) or CFrame.new())
	w.Parent = car.model
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = w; wc.Parent = w
	return w
end

local function wheels(car, x, zf, zr, r, w)
	for _, sx in ipairs({ -x, x }) do
		for _, z in ipairs({ zf, zr }) do
			cp(car, w, r * 2, r * 2, sx, r, z, Color3.fromRGB(22, 22, 24), Enum.Material.Rubber, { shape = Enum.PartType.Cylinder })
			local rim = cp(car, w + 0.02, r * 1.25, r * 1.25, sx + (sx > 0 and 0.01 or -0.01), r, z, Color3.fromRGB(175, 180, 186), Enum.Material.Metal, { shape = Enum.PartType.Cylinder, refl = 0.25 })
			rim.Name = "Rim"
			cp(car, w + 0.04, r * 0.4, r * 0.4, sx + (sx > 0 and 0.02 or -0.02), r, z, Color3.fromRGB(60, 62, 66), Enum.Material.Metal, { shape = Enum.PartType.Cylinder })
		end
	end
end

local function lights(car, w, zF, zR, yH, yT, isPlayer)
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yH, zF, Color3.fromRGB(255, 250, 225), Enum.Material.Neon)
		local t = cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yT, zR, Color3.fromRGB(255, 30, 20), Enum.Material.Neon)
		t.Name = "Tail"
		local sig = cp(car, 0.18, 0.1, 0.06, sx * (w * 0.5 - 0.12), yT, zR, AMBER_OFF, Enum.Material.Neon)
		local sigF = cp(car, 0.18, 0.1, 0.06, sx * (w * 0.5 - 0.12), yH, zF, AMBER_OFF, Enum.Material.Neon)
		if sx < 0 then car.sigL = { sig, sigF } else car.sigR = { sig, sigF } end
	end
end

function buildCar(kind, color)
	local car = { kind = kind, L = DIM[kind].L, W = DIM[kind].W, paint = color }
	car.model = Instance.new("Model"); car.model.Name = "Car_" .. kind
	car.root = mk(car.model, Vector3.new(0.4, 0.3, 0.4), CFrame.new(0, 300, 0), color, Enum.Material.SmoothPlastic, { tr = 1 })
	car.model.PrimaryPart = car.root
	BUILD[kind](car, color)
	car.model.Parent = world
	if kind == "vanta" or kind == "strada" or kind == "or1" then
		-- real headlight beams for the player's car
		local beam = Instance.new("SpotLight"); beam.Face = Enum.NormalId.Front; beam.Angle = 55; beam.Range = 60; beam.Brightness = 3
		beam.Color = Color3.fromRGB(255, 244, 220); beam.Shadows = true; beam.Parent = car.root
		local tail = Instance.new("PointLight"); tail.Range = 8; tail.Brightness = 1.2; tail.Color = Color3.fromRGB(255, 40, 30)
		local tp = car.model:FindFirstChild("Tail"); if tp then tail.Parent = tp end
	end
	return car
end

local function updateHills(f)
	for _, h in ipairs(hills) do
		local z = h.z - (f * 0.15) % 1920 -- slow parallax drift
		if z < -960 then z += 1920 end
		h.part.CFrame = CFrame.new(h.x * S, -6 * S, -(f + z + 700) * S)
	end
end
for i = 1, SEG_COUNT do segs[i] = { model = buildSegment(), k = i - 1 } end
local function placeSegment(s, k) s.k = k; s.model:PivotTo(CFrame.new(0, 0, -(k + 0.5) * SEG_LEN * S)) end
for i, s in ipairs(segs) do placeSegment(s, i - 4) end
function updateRoad(f)
	updateHills(f)
	local base = math.floor(f / SEG_LEN) - 3
	for i, s in ipairs(segs) do
		local k = base + ((i - 1 - base) % SEG_COUNT)
		if s.k ~= k then placeSegment(s, k) end
	end
end

-- traffic tuning follows the web version's endless curve: heavier and faster the farther you go
local function updateDiff()
	local D = curDiff()
	local t = math.min(1, P.f / (8 * MILE))
	local amt = clamp(SET.traffic, 0, 2)
	DIFF.traffic = math.floor((10 + t * 22) * 1.2 * D.traffic * amt + 0.5)
	DIFF.speed = (0.92 + t * 0.33) * D.speed
	DIFF.lane = math.min(0.9, (0.14 + t * 0.46) * D.lane)
end

local function laneFree(lane, f, except, gap)
	for _, o in ipairs(traffic) do
		if o ~= except and (o.lane == lane or (o.change and o.change.to == lane)) and math.abs(o.f - f) < gap + (o.L + 4.6) / 2 then return false end
	end
	return true
end
local function spawnTraffic(f, lane)
	local r = math.random()
	local kind = r < 0.5 and "sedan" or r < 0.75 and "suv" or r < 0.88 and "van" or "semi"
	lane = lane or math.random(1, 4)
	if kind == "semi" and lane == 1 then lane = math.random(2, 4) end
	if not laneFree(lane, f, nil, 30) then return nil end
	local hue = math.random()
	local col = kind == "semi" and Color3.fromHSV(hue, 0.55, 0.7) or Color3.fromHSV(hue, math.random() < 0.4 and 0.05 or 0.6, math.random() < 0.4 and 0.35 or 0.8)
	local car = buildCar(kind, col)
	car.f = f; car.lane = lane; car.x = laneX(lane)
	car.desired = ((kind == "semi" and rnd(17, 22) or rnd(19, 31)) + (5 - lane) * 1.4) * DIFF.speed
	car.v = car.desired; car.yawv = 0; car.side = sgn(f - P.f); car.timer = rnd(2, 6)
	traffic[#traffic + 1] = car
	return car
end
local function clearTraffic()
	for _, c in ipairs(traffic) do c.model:Destroy() end
	traffic = {}
end

local function slider(parent, lo, hi, step, get, set)
	local track = frame(parent, { Size = UDim2.new(1, 0, 0, 22), BackgroundTransparency = 1 })
	local rail = frame(track, { Size = UDim2.new(1, -16, 0, 6), Position = UDim2.new(0, 8, 0.5, -3), BackgroundTransparency = 0.82, BackgroundColor3 = C_PAPER })

	local knob = frame(track, { Size = UDim2.fromOffset(18, 18), AnchorPoint = Vector2.new(0.5, 0.5), BackgroundTransparency = 0, BackgroundColor3 = C_PAPER })
	corner(knob, 9)
	local function refresh()
		local t = (get() - lo) / (hi - lo)
		fill.Size = UDim2.fromScale(t, 1); knob.Position = UDim2.new(0, 8 + t * (rail.AbsoluteSize.X), 0.5, 0)
	end
	local function setFromX(x)
		local t = clamp((x - rail.AbsolutePosition.X) / math.max(1, rail.AbsoluteSize.X), 0, 1)
		local v = lo + math.floor((t * (hi - lo)) / step + 0.5) * step
		set(clamp(v, lo, hi)); refresh()
	end
	local hit = btn(track, { Size = UDim2.fromScale(1, 1), Text = "" })
	hit.MouseButton1Down:Connect(function(x) activeSlider = setFromX; setFromX(x) end)
	rail:GetPropertyChangedSignal("AbsoluteSize"):Connect(refresh)
	task.defer(refresh)
	return track, refresh
end

---------------------------------------------------------------- snapshots / rewind
local function snapshot()
	local s = { p = {}, cars = {}, ind = ind }
	for k, v in pairs(P) do s.p[k] = v end
	for _, c in ipairs(traffic) do
		s.cars[#s.cars + 1] = { c = c, f = c.f, x = c.x, v = c.v, lane = c.lane, crashed = c.crashed, yawv = c.yawv }
	end
	return s
end
-- Put the run back to a snapshot. Cars that did not exist yet are removed for real (not just hidden),
-- so nothing invisible is left on the road to crash into.
local function restore(s)
	for k, v in pairs(s.p) do P[k] = v end
	ind = s.ind
	local keep, list = {}, {}
	for _, e in ipairs(s.cars) do
		local c = e.c
		if c.model.Parent then
			c.f = e.f; c.x = e.x; c.v = e.v; c.lane = e.lane; c.crashed = e.crashed; c.yawv = e.yawv or 0
			c.change = nil; c.yb = false; c.yr = nil; c.side = sgn(c.f - P.f); c.counted = false
			keep[c] = true; list[#list + 1] = c
		end
	end
	for _, c in ipairs(traffic) do if not keep[c] then c.model:Destroy() end end
	traffic = list
end

local RAMP = { L = 11, W = 3.0, H = 1.7 }
local RAMP_X = ROAD_HALF - 0.4 + SHOULDER / 2 + 0.1
local ramps, pickups = {}, {}
local stuntFolder = Instance.new("Folder"); stuntFolder.Name = "Stunt"; stuntFolder.Parent = world
local nextRampF, nextPickF = 0, 0
local function clearStunt()
	for _, r in ipairs(ramps) do r.model:Destroy() end
	for _, p in ipairs(pickups) do p.part:Destroy() end
	ramps, pickups = {}, {}
end
local function addPickup(f, x, y)
	local part = mk(stuntFolder, Vector3.new(0.9, 0.45, 0.45), CFrame.new(), Color3.fromRGB(70, 170, 255), Enum.Material.Neon, { shape = Enum.PartType.Cylinder, shadow = false })

	pickups[#pickups + 1] = { f = f, x = x, y = y, part = part }
end
local function addRamp(f)
	local m = Instance.new("Model"); m.Name = "Ramp"
	local x = RAMP_X
	local w = Instance.new("WedgePart"); w.Anchored = true; w.CanCollide = false; w.CanQuery = false
	w.Size = Vector3.new(RAMP.W, RAMP.H, RAMP.L) * S; w.Color = Color3.fromRGB(245, 181, 36); w.Material = Enum.Material.DiamondPlate
	w.CFrame = CFrame.new(w2v(x, RAMP.H / 2, f)) * CFrame.Angles(0, math.pi, 0); w.Parent = m
	for _, sx in ipairs({ -1, 1 }) do
		mk(m, Vector3.new(0.2, 0.25, RAMP.L), CFrame.new(w2v(x + sx * (RAMP.W / 2 + 0.1), 0.12, f)), Color3.fromRGB(20, 20, 22), nil, { shadow = false })
	end
	for k = 1, 4 do -- chevrons on the shoulder leading in
		mk(m, Vector3.new(1.2, 0.03, 0.5), CFrame.new(w2v(x, 0.02, f - RAMP.L / 2 - 6 - k * 7)), Color3.fromRGB(245, 181, 36), Enum.Material.Neon, { shadow = false })
	end
	m.Parent = stuntFolder
	ramps[#ramps + 1] = { f = f, x = x, model = m }
	for k = 1, 3 do addPickup(f + RAMP.L / 2 + 12 + k * 9, x - k * 1.6, 3.2 + k * 0.4) end
end
local function stuntLand()
	local dist = P.f - P.airF
	P.jumps += 1; P.longest = math.max(P.longest, dist)
	local pts = math.floor((dist * 6 + P.cleared * 400) * paceMult(P.fwd * 2.23694))
	P.score += pts
	popup(("BIG AIR · %d m"):format(math.floor(dist)), (P.cleared > 0 and ("Cleared %d · "):format(P.cleared) or "") .. "+" .. commas(pts))
	play("land", 0.6, 0.7); shake = math.max(shake, 0.5)
	P.cleared = 0
end
local function updateStunt(dt)
	if SET.mode ~= "stunt" then return end
	if nextRampF < P.f + 140 then nextRampF = math.max(nextRampF, P.f + 240); addRamp(nextRampF); nextRampF += rnd(420, 700) end
	if nextPickF < P.f + 150 then nextPickF = math.max(nextPickF, P.f + 180); addPickup(nextPickF, laneX(math.random(1, 4)), 0.7); nextPickF += rnd(120, 220) end
	for i = #ramps, 1, -1 do if ramps[i].f < P.f - 150 or ramps[i].f > P.f + 900 then ramps[i].model:Destroy(); table.remove(ramps, i) end end
	local ground, onRamp = 0, false
	for _, r in ipairs(ramps) do
		local t = (P.f - (r.f - RAMP.L / 2)) / RAMP.L
		if math.abs(P.x - r.x) < RAMP.W / 2 + 0.3 and t >= 0 and t <= 1 then
			ground = t * RAMP.H; onRamp = true
			if not P.air and t > 0.97 then
				P.air = true; P.airF = P.f; P.cleared = 0; P.y = RAMP.H
				P.vy = P.v * (RAMP.H / RAMP.L) * 1.05 + 1.5
			end
		end
	end
	if P.air then
		P.vy -= 15 * dt; P.y += P.vy * dt
		if P.y <= 0 then P.y = 0; P.vy = 0; P.air = false; stuntLand() end
	elseif onRamp then P.y = ground else P.y = 0 end
	for i = #pickups, 1, -1 do
		local pk = pickups[i]
		if pk.f < P.f - 30 or pk.f > P.f + 900 then pk.part:Destroy(); table.remove(pickups, i)
		elseif math.abs(pk.f - P.f) < 2.6 and math.abs(pk.x - P.x) < 1.5 and math.abs(pk.y - (P.y + 0.6)) < 1.7 then
			P.nitro = math.min(1, P.nitro + 0.34); P.score += 250
			popup("NITRO", "+250"); play("pickup", 0.4, 1.4)
			pk.part:Destroy(); table.remove(pickups, i)
		else
			pk.part.CFrame = CFrame.new(w2v(pk.x, pk.y + math.sin(os.clock() * 3 + i) * 0.1, pk.f)) * CFrame.Angles(0, os.clock() * 2, math.rad(90))
		end
	end
end

---------------------------------------------------------------- crash, damage and close calls (rules ported from the web version)
local KIND = { sedan = "a sedan", suv = "an SUV", van = "a delivery van", semi = "a semi truck" }
local KIND_UP = { sedan = "SEDAN", suv = "SUV", van = "VAN", semi = "BIG RIG" }
local function cap(s) return s:sub(1, 1):upper() .. s:sub(2) end
local function crash(msg, other, extra)
	if P.crashed then return end
	P.crashed = true; P.crashT = 0; P.msg = msg
	local E = clamp((P.fwd + (extra or 0) * 0.8 - 15) / 75, 0, 1); P.crashE = E
	local vlat = math.sin(P.h) * P.v
	P.cvz = P.fwd * (other and 0.55 or 0.4)
	P.cvx = vlat * 0.4 + (other and (sgn(P.x - other.x) ~= 0 and sgn(P.x - other.x) or 1) * rnd(3, 6) * (1 + E) or -sgn(vlat) * 3)
	P.spin = (math.random() < 0.5 and -1 or 1) * rnd(2.5, 4.5) * (0.6 + E * 0.9)
	P.vy = E > 0.25 and 2.5 + E * 9 or 0; P.air = false
	P.rollV = E > 0.25 and (math.random() < 0.5 and -1 or 1) * (2.5 + E * 5) or 0
	P.pitchV = E > 0.45 and rnd(-1, 1) * E * 5 or 0
	P.cRoll, P.cPitch = 0, 0
	if other then
		other.crashed = true; other.spinV = -P.spin * 0.5; other.change = nil
		other.v = math.max(0, other.v + (P.fwd - other.v) * (0.35 + E * 0.25))
	end
	shake = 0.6 + E; play("crash", 0.6 + 0.4 * E, 0.8)
	local e = Instance.new("Explosion"); e.BlastPressure = 0; e.BlastRadius = 1; e.DestroyJointRadiusPercent = 0
	e.Position = playerCar.root.Position + Vector3.new(0, 3, 0); e.Visible = E > 0.4; e.Parent = Workspace
end
local function takeDamage(amount, c, what)
	P.damage = math.min(what == "scrape" and SET.swipeSafe and 0.99 or 1, P.damage + amount)
	if P.damage >= 1 then
		crash(("Your car couldn't take another %s. %s finished it off at %d mph."):format(what, cap(KIND[c.kind]), math.floor(P.fwd * 2.23694 + 0.5)), c)
		return
	end
	popup(what == "scrape" and "SIDE-SWIPE" or "BUMPER TAP", ("Damage %d%%"):format(math.floor(P.damage * 100 + 0.5)))
end
local function sideSwipe(c, dx, ox, toward)
	local s = sgn(dx); if s == 0 then s = 1 end
	P.x += s * (ox + 0.02)
	if sgn(P.h) == -s then P.h *= -0.3 end
	P.h = clamp(P.h + s * 0.025, -0.7, 0.7); P.yaw = 0; P.si *= 0.3
	P.v *= 0.99; c.counted = true
	if c.change and sgn(c.change.to - c.change.from) == -s then
		c.change = { from = c.change.to, to = c.change.from, t = 1 - clamp(c.change.t, 0, 1) }
	end
	shake = math.max(shake, 0.35)
	if (c.scrapeCD or 0) > P.clock then return end
	c.scrapeCD = P.clock + 0.5
	play("crash", 0.25, 1.6)
	takeDamage(curDiff().dmg * (0.7 + math.min(1, toward / 5) * 0.8), c, "scrape")
end
local function bumperTap(c, oz, behind)
	if behind then c.f -= oz + 0.05; c.v = math.min(c.v, P.fwd - 1.5); P.v += 0.8
	else c.f += oz + 0.05; P.v = math.max(0, math.min(P.v, c.v - 1.5)); c.v += 1 end
	c.counted = true; shake = math.max(shake, 0.45)
	if (c.scrapeCD or 0) > P.clock then return end
	c.scrapeCD = P.clock + 0.5
	play("crash", 0.35, 1.1)
	takeDamage(curDiff().dmg * 0.6, c, "bump")
end
local function nearMiss(c, clearance)
	if c.counted then return end
	c.counted = true
	P.combo = P.comboT > 0 and P.combo + 1 or 1; P.comboT = 4.5
	P.bestCombo = math.max(P.bestCombo, P.combo); P.near += 1
	local mph = P.fwd * 2.23694
	local side = sgn(c.x - P.x); if side == 0 then side = 1 end
	local thread = P.clock - P.lastNearT < 0.7 and side ~= P.lastNearSide
	P.lastNearT = P.clock; P.lastNearSide = side
	local pts = math.floor((150 + (1 - clearance) * 350) * paceMult(mph) * math.min(P.combo, 8) * (thread and 2 or 1) + 0.5)
	P.score += pts
	popup(thread and "THREADED THE GAP" or clearance < 0.35 and "PAINT SWAP" or "CLOSE CALL", ("%s · %d cm · +%s"):format(KIND_UP[c.kind], math.floor(clearance * 100), commas(pts)))
	comboScale.Scale = 1.15
	play("closeCall", 0.35, 0.9 + math.min(P.combo, 8) * 0.06)
end

---------------------------------------------------------------- player physics (ported from the web version's updatePlayer)
local function stepCrash(dt)
	P.crashT += dt
	local air = P.y > 0.001 or P.vy > 0
	P.h += P.spin * dt; P.spin *= math.exp(-(air and 0.3 or 1.1) * dt)
	P.cvx *= math.exp(-(air and 0.25 or 1.6) * dt)
	local upside = math.cos(P.cRoll) * math.cos(P.cPitch) < 0
	P.cvz = math.max(0, P.cvz - (air and 1.2 or upside and 6 or 9) * dt)
	P.x += P.cvx * dt
	local lim = RAIL_X - 1.4
	if P.x < -lim then P.x = -lim; P.cvx = math.abs(P.cvx) * 0.3 end
	if P.x > lim then P.x = lim; P.cvx = -math.abs(P.cvx) * 0.3 end
	if air then
		P.vy -= 9.8 * dt; P.y += P.vy * dt
		P.cRoll += P.rollV * dt; P.cPitch += P.pitchV * dt
		if P.y <= 0 then
			P.y = 0
			local impact = -P.vy
			if impact > 2.2 then
				P.vy = impact * 0.42; P.rollV *= 0.72; P.pitchV *= 0.6; P.cvz *= 0.88
				shake = math.max(shake, 0.3 + impact * 0.06); play("land", math.min(1, 0.25 + impact * 0.08), 0.8)
			else P.vy = 0 end
		end
	else
		P.cRoll = damp(P.cRoll, math.floor(P.cRoll / math.pi + 0.5) * math.pi, 6, dt)
		P.cPitch = damp(P.cPitch, math.floor(P.cPitch / math.pi + 0.5) * math.pi, 6, dt)
	end
	P.fwd = P.cvz; P.v = P.cvz; P.f += P.cvz * dt
	P.throttle = 0
end
local function stepPlayer(dt)
	if P.crashed then stepCrash(dt); return end
	local car = CUR
	local left, right = down(Enum.KeyCode.A, Enum.KeyCode.Left), down(Enum.KeyCode.D, Enum.KeyCode.Right)
	local gas, brake = down(Enum.KeyCode.W, Enum.KeyCode.Up), down(Enum.KeyCode.S, Enum.KeyCode.Down, Enum.KeyCode.Space)
	local airborne = P.air
	local steerIn = (right and 1 or 0) - (left and 1 or 0)
	local brk = brake and 1 or 0
	-- the virtual wheel turns at a limited rate, input is eased, and yaw follows with a little lag
	local sens = SET.steer
	local reversing = steerIn ~= 0 and P.si ~= 0 and sgn(steerIn) ~= sgn(P.si)
	local wheelRate = (steerIn == 0 and 3.2 + sens * 1.6 or reversing and 4.5 + sens * 2.5 or 1.6 + sens * 1.9) * car.steer
	P.si += clamp(steerIn - P.si, -wheelRate * dt, wheelRate * dt)
	local siEased = P.si * (0.35 + 0.65 * math.abs(P.si))
	P.throttle = damp(P.throttle, gas and 1 or 0, 8, dt)
	local v = math.max(P.v, 0.1)
	local wet = SET.weather == "rain" and 0.85 or 1
	local maxYaw = math.min(0.9 + sens * 0.45, ((6.2 + sens * 2.8) * car.grip * wet) / math.max(v, 4))
	-- with auto-straighten off, letting go holds the current heading
	local centring = SET.autoCenter and clamp(-P.h * (1.8 + sens * 1.0), -maxYaw, maxYaw) or 0
	local yawTarget = math.abs(P.si) > 0.02 and siEased * maxYaw or centring
	if airborne then yawTarget *= 0.35 end
	P.yaw = damp(P.yaw, yawTarget, (4.5 + sens * 3) * car.steer, dt)
	local lat, grip = P.yaw * v, (8.4 + sens * 1.6) * car.grip * wet
	if math.abs(lat) > grip then P.yaw *= grip / math.abs(lat) end
	P.h = clamp(P.h + P.yaw * dt, -0.7, 0.7)
	-- longitudinal
	local m, power = 1520, car.power
	local drive = airborne and 0 or P.throttle * math.min(m * (5.4 + power / 1.2e6), power / math.max(v, 3))
	local drag, rollR, engBrake = car.drag * v * v, m * 9.81 * 0.013, (1 - P.throttle) * m * 0.55
	local brakeF = (airborne and 0 or brk) * m * (12.5 + math.min(v, 80) * 0.035) * car.brake * wet
	local a = (drive - drag - rollR - engBrake - brakeF) / m - math.abs(lat) * 0.025
	P.boosting = SET.mode == "stunt" and P.nitro > 0 and down(Enum.KeyCode.LeftShift, Enum.KeyCode.RightShift, Enum.KeyCode.N)
	if P.boosting then a += 15; P.nitro = math.max(0, P.nitro - dt * 0.2) end
	P.v = clamp(P.v + a * dt, 0, P.boosting and car.top * 1.22 or math.max(car.top, P.v - 10 * dt))
	-- gears & rpm
	local rpm = (P.v / TIRE_C) * 60 * GEARS[P.gear] * car.final
	if rpm > 7400 and P.gear < #GEARS then P.gear += 1 elseif rpm < 3200 and P.gear > 1 then P.gear -= 1 end
	rpm = math.max(900 + P.throttle * 600, (P.v / TIRE_C) * 60 * GEARS[P.gear] * car.final)
	P.rpm = damp(P.rpm, math.min(rpm, 7900), 12, dt)
	-- integrate
	P.x += math.sin(P.h) * P.v * dt
	P.fwd = math.cos(P.h) * P.v
	P.f += P.fwd * dt
	P.top = math.max(P.top, P.fwd)
	-- guardrails: glance off at shallow angles, wreck if you drive straight into one
	local hw = (playerCar.W / 2) * math.cos(P.h) + (playerCar.L / 2) * math.abs(math.sin(P.h))
	local function hitWall()
		local vlat = math.sin(P.h) * P.v
		if math.abs(vlat) > 10 then crash(("You hit the guardrail at %d mph."):format(math.floor(P.fwd * 2.23694 + 0.5))); return end
		P.h = -P.h * 0.25; P.yaw = 0; P.v *= 0.6 ^ (dt * 6); shake = math.max(shake, 0.25)
		play("crash", 0.12, 1.8)
	end
	if P.x - hw < -RAIL_X then P.x = -RAIL_X + hw; hitWall()
	elseif P.x + hw > RAIL_X then P.x = RAIL_X - hw; hitWall() end
end

local function collisions()
	if P.crashed then return end
	local hw = (playerCar.W / 2) * math.cos(P.h) + (playerCar.L / 2) * math.abs(math.sin(P.h))
	local hl = (playerCar.L / 2) * math.cos(P.h) + (playerCar.W / 2) * math.abs(math.sin(P.h))
	local vlat = math.sin(P.h) * P.v
	for _, c in ipairs(traffic) do
		local dz = c.f - P.f -- + ahead
		local cwid = c.W / 2 + math.abs(math.sin(c.yawv or 0)) * c.L / 2
		local dx = P.x - c.x
		local ox, oz = hw + cwid - 0.05 - math.abs(dx), hl + c.L / 2 - 0.05 - math.abs(dz)
		local clearH = c.kind == "semi" and 4.1 or (c.kind == "van" and 2.3 or 1.55)
		if ox > 0 and oz > 0 and P.y > clearH then
			if not (P.clock - (c.jumpT or -9) < 3) then c.jumpT = P.clock; P.cleared += 1 end
			continue
		end
		if ox > 0 and oz > 0 and not c.crashed then
			local toward = math.max(0, -sgn(dx) * vlat)
			if ox < oz and ox < 0.55 and toward < 7 then sideSwipe(c, dx, ox, toward); continue end
			if ox >= oz and oz < 0.9 and math.abs(P.fwd - c.v) < 4.5 then bumperTap(c, oz, dz < 0); continue end
		end
		if ox > 0 and oz > 0 then
			local rear = dz < 0 and down(Enum.KeyCode.S, Enum.KeyCode.Down, Enum.KeyCode.Space)
			local mph = math.floor(P.fwd * 2.23694 + 0.5)
			crash(rear and ("%s rear-ended you at %d mph."):format(cap(KIND[c.kind]), mph) or ("You clipped %s at %d mph."):format(KIND[c.kind], mph), c)
			return
		end
		-- close call: the player's front passes the car's rear (or the reverse)
		local side = dz < 0 and -1 or 1
		if side ~= c.side and math.abs(dz) < c.L / 2 + 2.6 then
			local hw2 = playerCar.W / 2 + (playerCar.L / 2) * math.abs(math.sin(P.h))
			local clearance = math.abs(c.x - P.x) - hw2 - c.W / 2
			if clearance < 1.0 and clearance > -0.1 and P.fwd > 18 then nearMiss(c, clearance) end
		end
		c.side = side
	end
end

---------------------------------------------------------------- traffic
local function stepTraffic(dt)
	updateDiff()
	for i = #traffic, 1, -1 do
		local c = traffic[i]
		local target = c.crashed and 0 or c.desired
		if not c.crashed then
			for _, o in ipairs(traffic) do
				if o ~= c and math.abs(o.x - c.x) < 2.4 and o.f > c.f then
					local gap = o.f - c.f - (o.L + c.L) / 2
					if gap < 8 + c.v * 1.1 then target = math.min(target, o.v - (8 + c.v - gap) * 0.05) end
				end
			end
			-- don't drive into the player's car either
			if math.abs(P.x - c.x) < 2.2 and P.f > c.f and not P.crashed then
				local gap = P.f - c.f - (playerCar.L + c.L) / 2
				if gap < 6 + c.v * 0.6 then target = math.min(target, P.fwd - 1) end
			end
			if c.yb then target = math.min(target, math.max(0, P.v * 0.72)) end
		end
		c.v = math.max(0, c.v + clamp(target - c.v, -9 * dt, 3 * dt))
		c.f += c.v * dt
		if c.crashed then c.yawv = (c.yawv or 0) + (c.spinV or 0) * dt; c.spinV = (c.spinV or 0) * math.exp(-1.5 * dt) end
		-- random lane changes, more often on harder settings and farther along
		if not c.crashed and not c.change then
			c.timer -= dt
			if c.timer <= 0 then
				c.timer = rnd(3, 8)
				if math.random() < DIFF.lane then
					local to = c.lane + (math.random() < 0.5 and -1 or 1)
					if to >= 1 and to <= 4 and not (c.kind == "semi" and to == 1) and laneFree(to, c.f, c, 10 + c.v * 0.5) then
						c.change = { from = c.lane, to = to, t = -0.4 }
					end
				end
			end
		end
		if c.change then
			c.change.t += dt / (c.change.fast and 1.8 or 2.8)
			local t = clamp(c.change.t, 0, 1)
			local e = t * t * (3 - 2 * t)
			local nx = laneX(c.change.from) + (laneX(c.change.to) - laneX(c.change.from)) * e
			if dt > 0 then c.yawv = c.change.t > 0 and -(nx - c.x) / math.max(dt, 1e-3) / math.max(c.v, 5) or 0 end
			c.x = nx
			if c.change.t >= 1 then c.lane = c.change.to; c.x = laneX(c.lane); c.change = nil; c.yawv = 0 end
		end
		if c.f < P.f - 160 or c.f > P.f + 520 then c.model:Destroy(); table.remove(traffic, i) end
	end
	local tries = 0
	while #traffic < DIFF.traffic and tries < 6 do spawnTraffic(P.f + rnd(260, 440)); tries += 1 end
	while #traffic > DIFF.traffic + 4 do -- slider turned down: drop the farthest car ahead
		local far, fi = -1e9, nil
		for i, c in ipairs(traffic) do if c.f > far and c.f > P.f + 150 then far = c.f; fi = i end end
		if not fi then break end
		traffic[fi].model:Destroy(); table.remove(traffic, fi)
	end
end

local function yieldToIndicator(dt)
	if ind == 0 or state ~= "play" or P.crashed then return end
	indT += dt
	local pl, bd = 2, 99
	for i = 1, 4 do local d = math.abs(P.x - laneX(i)); if d < bd then bd = d; pl = i end end
	if bd > 2.6 then return end
	local tl = pl + ind
	if tl < 1 or tl > 4 then return end
	for _, c in ipairs(traffic) do
		if c.lane == tl and not c.change and not c.crashed and c.f - P.f > -30 and c.f - P.f < 60 then
			c.yr = c.yr or (indT + rnd(0.4, 1.3))
			if indT >= c.yr then
				local fl = tl + ind
				if fl >= 1 and fl <= 4 and laneFree(fl, c.f, c, 12 + c.v * 0.5) then c.change = { from = c.lane, to = fl, t = -0.25, fast = true }; c.yb = false
				else c.yb = c.f - P.f < 8 end
			end
		elseif c.yb and c.lane ~= tl then c.yb = false end
	end
end

local function bank()

	local m = SET.mode
	local newBest = sc > (PROG.best[m] or 0)

		PROG.career += sc
		if newBest then PROG.best[m] = sc end
		local ev = RS:FindFirstChild("IWSubmit"); if ev then ev:FireServer(m, sc) end
	end
	return sc, newBest
end

end
restartRun = function()
	if state ~= "play" and state ~= "paused" and state ~= "over" then return end
	-- keep what this run earned before throwing it away
	local sc = 0
	if state ~= "over" then sc = bank() end
	startGame()
	if sc > 0 then popup("RUN BANKED", "+" .. commas(sc) .. " career pts kept") end
end
togglePause = function()
	if state == "play" then state = "paused"; pauseO.Visible = true
	elseif state == "paused" then state = "play"; pauseO.Visible = false; setO.Visible = false end
end
local function startRewind()
	if #hist < 4 or rewinding then return end
	if state == "over" then over.Visible = false; state = "play" end
	if SET.rewindFx then
		rewinding = { i = #hist, acc = 0, n = #hist }
		rewFx.Visible = true; play("tick", 0.4, 0.6)
	else
		restore(hist[1]); for k = #hist, 2, -1 do hist[k] = nil end
		P.crashed = false; popup("REWIND", "Back 7 seconds"); doFlash(0.5)
	end
end

---------------------------------------------------------------- input
UIS.InputBegan:Connect(function(input, gp)
	if gp then return end
	keys[input.KeyCode] = true
	local k = input.KeyCode
	if setO.Visible then if k == Enum.KeyCode.Escape then setO.Visible = false end return end
	if k == Enum.KeyCode.R and (state == "play" or state == "over") then startRewind()
	elseif k == Enum.KeyCode.C and state == "play" then camMode = 1 - camMode
	elseif (k == Enum.KeyCode.P or k == Enum.KeyCode.Escape) and (state == "play" or state == "paused") then togglePause()
	elseif k == Enum.KeyCode.M then SET.muted = not SET.muted; popup(SET.muted and "SOUND OFF" or "SOUND ON"); saveSettings()
	elseif k == Enum.KeyCode.Q and state == "play" then ind = (ind == -1) and 0 or -1; indT = 0; for _, c in ipairs(traffic) do c.yr = nil; c.yb = false end
	elseif k == Enum.KeyCode.E and state == "play" then ind = (ind == 1) and 0 or 1; indT = 0; for _, c in ipairs(traffic) do c.yr = nil; c.yb = false end
	elseif k == Enum.KeyCode.Return and (state == "menu" or state == "over") then startGame()
	end
end)
UIS.InputEnded:Connect(function(input) keys[input.KeyCode] = nil end)
UIS.WindowFocusReleased:Connect(function() keys = {}; if state == "play" then togglePause() end end)

---------------------------------------------------------------- main loop
RunService.RenderStepped:Connect(function(dtRaw)
	local dt = math.min(dtRaw, 0.05)
	camera.CameraType = Enum.CameraType.Scriptable
	VOL.sfx = SET.sfx; VOL.muted = SET.muted
	if not playerCar then showCar() end
	if state == "menu" then
		poseCar(playerCar, P.x, 0, 0, 0, 0)
		updateRoad(0); updateCamera(dt)
		rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 26, -12)); rain.Rate = rainOn and 1400 or 0
		blurFx.Size = 0
		return
	end
	if state == "paused" then return end

	if rewinding then
		-- scrub backwards through the snapshots like a tape, easing in and out
		rewinding.acc += dt / (0.5 + 0.08 * rewinding.n / 10)
		local x = math.min(1, rewinding.acc)
		local e = x < 0.5 and 2 * x * x or 1 - (-2 * x + 2) ^ 2 / 2
		local idx = math.max(1, math.floor((1 - e) * (rewinding.n - 1) + 1.5))
		restore(hist[idx])
		rewSec.Text = ("-%.1f s"):format((rewinding.n - 1) / 10 * e)
		scan.Position = UDim2.fromScale(0, (os.clock() * 1.8) % 1.2 - 0.14)
		if x >= 1 then
			for k = #hist, 2, -1 do hist[k] = nil end
			restore(hist[1]); histT = 0
			rewinding = nil; rewFx.Visible = false; P.crashed = false; camPos = nil
			doFlash(0.5); popup("REWIND", (#hist >= 1 and "Back 7 seconds" or ""))
		end
	elseif state == "play" or state == "over" then
		P.clock += dt
		stepPlayer(dt)
		if not P.crashed then updateStunt(dt) end
		yieldToIndicator(dt); stepTraffic(dt); collisions()
		if not P.crashed then
			P.score += P.fwd * dt * 0.1 * paceMult(P.fwd * 2.23694)
			P.comboT = math.max(0, P.comboT - dt)
			histT += dt
			if histT >= 0.1 then
				histT -= 0.1; hist[#hist + 1] = snapshot()
				if #hist > REWIND_S * 10 + 1 then table.remove(hist, 1) end
			end
		elseif state == "play" and P.crashT > 2.2 then finishGame() end
	end
	flash.BackgroundTransparency = math.min(1, flash.BackgroundTransparency + dt * 2)

	updateRoad(P.f)
	poseCar(playerCar, P.x, P.f, -P.h, P.crashed and P.cRoll or -P.yaw * 0.06, P.y, P.crashed and P.cPitch or (P.air and clamp(P.vy * 0.03, -0.3, 0.3) or 0))
	for _, c in ipairs(traffic) do poseCar(c, c.x, c.f, c.yawv or 0, 0, 0) end
	updateIndicators(); updateCamera(dt); updateHud(dt)
	-- motion blur at speed (Roblox has no radial blur, so this is a gentle full-screen blur)
	local lvl = SET.motionBlur == "strong" and 1 or SET.motionBlur == "light" and 0.5 or 0
	local xs = clamp((P.fwd - 26) / 70, 0, 1)
	blurFx.Size = (state == "play" and not P.crashed) and lvl * 5 * xs * xs * (3 - 2 * xs) + (P.boosting and lvl * 2 or 0) or 0
	rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 26, -12)) * CFrame.Angles(-math.atan2(P.v * S, 150) * 0.6, 0, 0)
	rain.Rate = rainOn and 1400 or 0
end)

---------------------------------------------------------------- load saved settings and progress
task.spawn(function()
	local fn = RS:WaitForChild("IWLoad", 10)
	if not fn then return end
	local ok, res = pcall(function() return fn:InvokeServer() end)
	if not ok or type(res) ~= "table" then return end
	if type(res.settings) == "table" then
		for k, v in pairs(res.settings) do if SET[k] ~= nil and type(v) == type(SET[k]) then SET[k] = v end end
	end
	if not PAINTS[SET.paint] then SET.paint = 1 end
	if type(res.career) == "number" then PROG.career = math.max(PROG.career, res.career) end
	if type(res.best) == "table" then for m, v in pairs(res.best) do if type(v) == "number" then PROG.best[m] = math.max(PROG.best[m] or 0, v) end end end
	CUR = carDef(SET.car); if not unlocked(CUR) then SET.car = "vanta"; CUR = carDef("vanta") end
	applyTod(); if state == "menu" then showCar(); refreshMenu() end
end)
