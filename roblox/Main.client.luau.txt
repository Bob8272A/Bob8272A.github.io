local function mk(parent, size, cf, color, mat, o)
	local p = Instance.new("Part")
	p.Anchored = true; p.CanCollide = false; p.CanTouch = false; p.CanQuery = false
	p.Size = size * S; p.CFrame = cf; p.Color = color
	p.Material = mat or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth; p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		if o.tr then p.Transparency = o.tr end
		if o.refl then p.Reflectance = o.refl end
		if o.shape then p.Shape = o.shape end
		if o.shadow == false then p.CastShadow = false end
	end
	p.Parent = parent
	return p
end

-- part welded to a car root (offsets in metres, optional X tilt in degrees)
local function cp(car, sx, sy, sz, ox, oy, oz, color, mat, o)
	local rot = CFrame.new()
	if o and o.rx then rot = CFrame.Angles(math.rad(o.rx), 0, 0) end
	local p = mk(car.model, Vector3.new(sx, sy, sz), car.root.CFrame * CFrame.new(ox * S, oy * S, oz * S) * rot, color, mat, o)
	p.Anchored = false; p.Massless = true
	local wc = Instance.new("WeldConstraint"); wc.Part0 = car.root; wc.Part1 = p; wc.Parent = p
	return p
end

local GLASS = Color3.fromRGB(18, 28, 40)
local BLACK = Color3.fromRGB(14, 15, 17)
local AMBER_OFF, AMBER_ON = Color3.fromRGB(70, 42, 8), Color3.fromRGB(255, 160, 30)

local function wheels(car, x, zf, zr, r, w)
	for _, sx in ipairs({ -x, x }) do
		for _, z in ipairs({ zf, zr }) do
			local p = cp(car, w, r * 2, r * 2, sx, r, z, BLACK, Enum.Material.SmoothPlastic, { shape = Enum.PartType.Cylinder })
			local rim = cp(car, w + 0.02, r * 1.1, r * 1.1, sx + (sx > 0 and 0.01 or -0.01), r, z, Color3.fromRGB(150, 155, 160), Enum.Material.Metal, { shape = Enum.PartType.Cylinder })
			rim.Name = "Rim"
		end
	end
end

local function lights(car, w, zF, zR, yH, yT, isPlayer)
	for _, sx in ipairs({ -1, 1 }) do
		cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yH, zF, Color3.fromRGB(255, 250, 225), Enum.Material.Neon)
		local t = cp(car, w * 0.22, 0.13, 0.06, sx * w * 0.32, yT, zR, Color3.fromRGB(255, 30, 20), Enum.Material.Neon)
		t.Name = "Tail"
		local sig = cp(car, 0.18, 0.1, 0.06, sx * (w * 0.5 - 0.12), yT, zR, AMBER_OFF, Enum.Material.Neon)
		local sigF = cp(car, 0.18, 0.1, 0.06, sx * (w * 0.5 - 0.12), yH, zF, AMBER_OFF, Enum.Material.Neon)
		if sx < 0 then car.sigL = { sig, sigF } else car.sigR = { sig, sigF } end
	end
end

local function buildSegment()
	local m = Instance.new("Model"); m.Name = "Segment"
	local L = SEG_LEN
	local road = mk(m, Vector3.new(ROAD_HALF * 2 + 6, 0.2, L), CFrame.new(0, -0.1 * S, 0), Color3.fromRGB(46, 47, 50), Enum.Material.Asphalt)
	road.Name = "Road"; roadParts[#roadParts + 1] = road
	m.PrimaryPart = road
	mk(m, Vector3.new(300, 0.3, L), CFrame.new(0, -0.35 * S, 0), Color3.fromRGB(70, 82, 56), Enum.Material.Grass)
	for _, sx in ipairs({ -1, 1 }) do
		mk(m, Vector3.new(0.18, 0.02, L), CFrame.new(sx * (ROAD_HALF - 0.4) * S, 0.005 * S, 0), Color3.fromRGB(235, 235, 225), nil, { shadow = false })
		mk(m, Vector3.new(0.14, 0.7, L), CFrame.new(sx * (ROAD_HALF + 2.2) * S, 0.35 * S, 0), Color3.fromRGB(120, 124, 128), Enum.Material.Metal)
	end
	for i = 1, 3 do
		for d = 0, 3 do
			mk(m, Vector3.new(0.14, 0.02, 3.2), CFrame.new((i - 2) * LANE_W * S, 0.005 * S, (-L / 2 + 5 + d * (L / 4)) * S), Color3.fromRGB(235, 235, 225), nil, { shadow = false })
		end
	end
	local lx = (rng:NextInteger(0, 1) == 0 and -1 or 1) * (ROAD_HALF + 3.4)
	mk(m, Vector3.new(0.3, 9, 0.3), CFrame.new(lx * S, 4.5 * S, 10 * S), Color3.fromRGB(70, 72, 76), Enum.Material.Metal)
	mk(m, Vector3.new(0.3, 0.3, 2.6), CFrame.new((lx - math.sign(lx) * 1.1) * S, 9.1 * S, 10 * S), Color3.fromRGB(70, 72, 76), Enum.Material.Metal)
	local head = mk(m, Vector3.new(0.5, 0.2, 0.9), CFrame.new((lx - math.sign(lx) * 2.1) * S, 9 * S, 10 * S), Color3.fromRGB(255, 214, 150), Enum.Material.Neon)
	local pl = Instance.new("PointLight"); pl.Range = 38; pl.Brightness = 1.6; pl.Color = Color3.fromRGB(255, 200, 140); pl.Parent = head

		local side = rng:NextInteger(0, 1) == 0 and -1 or 1
		local tx, tz, h = side * rng:NextNumber(ROAD_HALF + 6, ROAD_HALF + 45), rng:NextNumber(-L / 2, L / 2), rng:NextNumber(5, 9)
		mk(m, Vector3.new(0.5, h * 0.7, 0.5), CFrame.new(tx * S, h * 0.35 * S, tz * S), Color3.fromRGB(70, 50, 36), Enum.Material.Wood)

	end
	m.Parent = world
	return m
end
for i = 1, SEG_COUNT do segs[i] = { model = buildSegment(), k = i - 1 } end
local function placeSegment(s, k) s.k = k; s.model:PivotTo(CFrame.new(0, 0, -(k + 0.5) * SEG_LEN * S)) end
for i, s in ipairs(segs) do placeSegment(s, i - 4) end
local function updateRoad(f)
	local cur = math.floor(f / SEG_LEN)
	for _, s in ipairs(segs) do
		if s.k < cur - 3 then placeSegment(s, s.k + SEG_COUNT) end
	end
end

local function spawnTraffic(f, lane)
	local r = math.random()
	local kind = r < 0.5 and "sedan" or r < 0.75 and "suv" or r < 0.88 and "van" or "semi"
	lane = lane or math.random(1, 4)
	for _, c in ipairs(traffic) do
		if c.x == laneX(lane) and math.abs(c.f - f) < 34 then return nil end
		if c.change and c.change.to == lane and math.abs(c.f - f) < 34 then return nil end
	end
	local hue = math.random()
	local col = kind == "semi" and Color3.fromHSV(hue, 0.55, 0.7) or Color3.fromHSV(hue, math.random() < 0.4 and 0.05 or 0.6, math.random() < 0.4 and 0.35 or 0.8)
	local car = buildCar(kind, col)
	car.f = f; car.lane = lane; car.x = laneX(lane)
	car.desired = (kind == "semi" and rnd(17, 22) or rnd(19, 31)) + (5 - lane) * 1.4
	car.v = car.desired; car.rel = f - P.f; car.yawv = 0
	traffic[#traffic + 1] = car
	return car
end

local function clearTraffic()
	for _, c in ipairs(traffic) do c.model:Destroy() end
	traffic = {}
end

---------------------------------------------------------------- snapshots / rewind
local function snapshot()
	local s = { p = {}, cars = {}, ind = ind }
	for k, v in pairs(P) do s.p[k] = v end
	for _, c in ipairs(traffic) do s.cars[#s.cars + 1] = { c = c, f = c.f, x = c.x, v = c.v, lane = c.lane } end
	return s
end
local function restore(s)
	for k, v in pairs(s.p) do P[k] = v end
	ind = s.ind
	local alive = {}
	for _, e in ipairs(s.cars) do e.c.f = e.f; e.c.x = e.x; e.c.v = e.v; e.c.lane = e.lane; e.c.change = nil; alive[e.c] = true end
	for _, c in ipairs(traffic) do c.model.Parent = alive[c] and world or nil end
end

---------------------------------------------------------------- simulation
local function stepPlayer(dt)
	local wet = rainOn and 0.85 or 1
	local throttle = (down(Enum.KeyCode.W, Enum.KeyCode.Up) and 1 or 0)
	local brake = (down(Enum.KeyCode.S, Enum.KeyCode.Down) and 1 or 0)
	local str = (down(Enum.KeyCode.D, Enum.KeyCode.Right) and 1 or 0) - (down(Enum.KeyCode.A, Enum.KeyCode.Left) and 1 or 0)
	if P.crashed then throttle, brake, str = 0, 1, 0 end
	P.steer = P.steer + clamp(str - P.steer, -dt * 5, dt * 5)
	local drag = 0.9 + 0.00045 * P.v * P.v
	local a = throttle * P.acc * math.max(0, 1 - P.v / P.top) ^ 0.7 - drag - brake * 26 * wet * (P.crashed and 0.6 or 1)
	P.v = math.max(0, P.v + a * dt)
	local lat = 10.5 / (1 + P.v / 38) * wet
	P.vx = P.vx + (P.steer * lat - P.vx) * math.min(1, dt * 8)
	P.x = P.x + P.vx * dt
	P.yaw = -math.atan(P.vx / math.max(P.v, 6))
	P.f = P.f + P.v * dt
	local edge = ROAD_HALF - 0.4 - 1.0
	if math.abs(P.x) > edge then
		P.x = clamp(P.x, -edge, edge); P.vx = 0
		P.damage = P.damage + dt * 0.5; P.v = P.v * (1 - dt * 0.6)
		if P.damage >= 1 and not P.crashed then wreck("WRECKED") end
	end
end

local function stepTraffic(dt)
	for i = #traffic, 1, -1 do
		local c = traffic[i]
		-- car following
		local target = c.desired
		for _, o in ipairs(traffic) do
			if o ~= c and math.abs(o.x - c.x) < 2.4 and o.f > c.f then
				local gap = o.f - c.f - (o.L + c.L) / 2
				if gap < 8 + c.v * 1.1 then target = math.min(target, o.v - (8 + c.v - gap) * 0.05) end
			end
		end
		if c.yb then target = math.min(target, math.max(0, P.v * 0.72)) end
		c.v = c.v + clamp(target - c.v, -9 * dt, 3 * dt)
		c.f = c.f + c.v * dt
		-- lane change progress
		if c.change then
			c.change.t = c.change.t + dt / (c.change.fast and 1.8 or 2.8)
			local t = clamp(c.change.t, 0, 1)
			local e = t * t * (3 - 2 * t)
			local nx = laneX(c.change.from) + (laneX(c.change.to) - laneX(c.change.from)) * e
			c.yawv = -(nx - c.x) / math.max(dt, 1e-3) / math.max(c.v, 5)
			c.x = nx
			if c.change.t >= 1 then c.lane = c.change.to; c.x = laneX(c.lane); c.change = nil; c.yawv = 0 end
		end
		-- cull / respawn
		if c.f < P.f - 160 or c.f > P.f + 900 then
			c.model:Destroy(); table.remove(traffic, i)
		end
	end
	-- keep the road busy
	local want = 40
	local tries = 0
	while #traffic < want and tries < 6 do spawnTraffic(P.f + rnd(500, 850)); tries += 1 end
end

local function yieldToIndicator(dt)
	if ind == 0 or state ~= "play" or P.crashed then return end
	indT += dt
	local pl, bd = 2, 99
	for i = 1, 4 do local d = math.abs(P.x - laneX(i)); if d < bd then bd = d; pl = i end end
	if bd > 2.6 then return end
	local tl = pl + ind
	if tl < 1 or tl > 4 then return end
	for _, c in ipairs(traffic) do
		if c.lane == tl and not c.change and c.f - P.f > -30 and c.f - P.f < 60 then
			c.yr = c.yr or (indT + rnd(0.4, 1.3))
			if indT >= c.yr then
				local fl = tl + ind
				local free = fl >= 1 and fl <= 4
				if free then
					for _, o in ipairs(traffic) do
						if o ~= c and (o.lane == fl or (o.change and o.change.to == fl)) and math.abs(o.f - c.f) < 12 + c.v * 0.5 then free = false; break end
					end
				end
				if free then c.change = { from = c.lane, to = fl, t = -0.25, fast = true }; c.yb = false
				else c.yb = c.f - P.f < 8 end
			end
		elseif c.yb and c.lane ~= tl then c.yb = false end
	end
end

local function collisions(dt)
	if P.crashed then return end
	for _, c in ipairs(traffic) do
		local df = c.f - P.f
		local reach = (c.L + playerCar.L) / 2
		local dx = c.x - P.x
		local wr = (c.W + playerCar.W) / 2
		if math.abs(df) < reach and math.abs(dx) < wr then
			local overlap = wr - math.abs(dx)
			if overlap < 0.55 then
				-- side swipe
				local push = (dx >= 0 and -1 or 1)
				P.x = P.x + push * overlap * 0.5; P.vx = push * 1.5
				P.v = P.v * (1 - dt * 0.5)
				P.damage = P.damage + dt * 0.55; P.combo = 1; shake = math.max(shake, 0.4)
				if P.damage >= 1 then wreck("WRECKED") end
			else
				P.damage = 1; wreck(P.v - c.v > 25 and "TOTALLED" or "WRECKED")
			end
			return
		end
		-- close calls
		local rel = df
		if c.rel ~= nil and ((c.rel > 0 and rel <= 0) or (c.rel < 0 and rel >= 0)) and math.abs(dx) < LANE_W * 1.1 and math.abs(dx) > (c.W + playerCar.W) / 2 - 0.05 and P.v > 15 then
			P.comboT = 3; P.combo = math.min(8, P.combo + 1); P.near += 1
			local pts = 50 * P.combo
			P.score += pts; say(("CLOSE CALL +%d  x%d"):format(pts, P.combo), Color3.fromRGB(255, 243, 176))
		end
		c.rel = rel
	end
end

local function poseCar(car, x, f, yaw, extraRoll, y)
	car.root.CFrame = CFrame.new(w2v(x, y or 0, f)) * CFrame.Angles(0, yaw, extraRoll or 0)

local function blinkNow() return (os.clock() % 0.7) < 0.38 end

local function updateIndicators()
	local on = blinkNow()
	local function set(parts, lit) if parts then for _, p in ipairs(parts) do p.Color = lit and AMBER_ON or AMBER_OFF end end end
	set(playerCar.sigL, ind == -1 and on); set(playerCar.sigR, ind == 1 and on)
	indL.TextColor3 = (ind == -1 and on) and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(60, 50, 30)
	indR.TextColor3 = (ind == 1 and on) and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(60, 50, 30)
	for _, c in ipairs(traffic) do
		if c.change then
			local d = c.change.to - c.change.from
			if not c.sigL then
				-- traffic cars share the same builder; they only blink while changing lanes
			end
			if c.sigL then set(c.sigL, d < 0 and on); set(c.sigR, d > 0 and on) end
		elseif c.sigL then set(c.sigL, false); set(c.sigR, false) end
	end
end

local function updateCamera(dt)
	local car = playerCar
	local pos = car.root.Position
	local speedFov = clamp(P.v * 0.2, 0, 30)
	if camMode == 0 or state == "menu" then
		if state == "menu" then
			local a = os.clock() * 0.35
			camera.CFrame = CFrame.lookAt(pos + Vector3.new(math.sin(a) * 26, 8, math.cos(a) * 26), pos + Vector3.new(0, 2, 0))
			camera.FieldOfView = 55; setCarVisible(car, true)
			if cockpit.folder then cockpit.folder.Parent = nil end
		else
			local back = (16 + P.v * 0.05) * 1
			local want = pos + Vector3.new(P.x * 0.12 * S * -0.0 , 6.5, back * 1.0) + Vector3.new(-P.vx * 0.9, 0, 0)
			camPos = camPos and camPos:Lerp(want, math.min(1, dt * 7)) or want
			local look = pos + Vector3.new(0, 2.5, -22)
			local c = CFrame.lookAt(camPos, look)
			if shake > 0 then c = c * CFrame.new((math.random() - 0.5) * shake, (math.random() - 0.5) * shake, 0); shake = math.max(0, shake - dt * 2) end
			camera.CFrame = c
			camera.FieldOfView = 66 + speedFov
			setCarVisible(car, true)
			if cockpit.folder then cockpit.folder.Parent = nil end
		end
	else
		local eye = car.root.CFrame * CFrame.new(-0.4 * S, 1.2 * S, -0.1 * S)
		local c = eye
		if shake > 0 then c = c * CFrame.new((math.random() - 0.5) * shake * 0.5, (math.random() - 0.5) * shake * 0.5, 0); shake = math.max(0, shake - dt * 2) end
		camera.CFrame = c; camera.FieldOfView = 74 + speedFov * 0.6
		setCarVisible(car, false)
		layoutCockpit(c)
		cockpit.folder.Parent = world
	end
end

end

		rewTint.BackgroundTransparency = 0.82
		if rewinding.i <= 1 then
			for k = #hist, rewinding.i + 1, -1 do hist[k] = nil end
			rewinding = nil; rewLbl.Visible = false; rewTint.BackgroundTransparency = 1; P.crashed = false
			say("REWIND", Color3.fromRGB(190, 225, 255))
		end
	else
		if P.crashed then
			P.crashT += dt; stepPlayer(dt); stepTraffic(dt)
			if P.crashT > 1.4 and state == "play" then finishGame() end
		else
			stepPlayer(dt); yieldToIndicator(dt); stepTraffic(dt); collisions(dt)
			P.score += P.v * dt * 0.1
			if P.comboT > 0 then P.comboT -= dt; if P.comboT <= 0 then P.combo = 1 end end
			-- history every 0.1 s
			histT += dt
			if histT >= 0.1 then
				histT -= 0.1; hist[#hist + 1] = snapshot()
				if #hist > REWIND_S * 10 + 1 then table.remove(hist, 1) end
			end
		end
	end

	updateRoad(P.f)
	poseCar(playerCar, P.x, P.f, P.yaw, P.crashed and math.min(0.6, (P.crashT or 0) * 0.5) or 0)
	for _, c in ipairs(traffic) do poseCar(c, c.x, c.f, c.yawv or 0, 0) end
	updateIndicators()
	updateCamera(dt)
	updateHud(dt)
	rainPart.CFrame = CFrame.new(camera.CFrame.Position + Vector3.new(0, 38, 0)) * CFrame.Angles(math.atan2(P.v * S, 150), 0, 0)
	rain.Rate = rainOn and 900 or 0
	rain.Speed = NumberRange.new(math.sqrt(150 ^ 2 + (P.v * S) ^ 2) - 10, math.sqrt(150 ^ 2 + (P.v * S) ^ 2) + 10)
	local headCar = playerCar
end)

toMenu()
