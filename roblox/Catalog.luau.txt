local byId, catById = {}, {}
for _, p in ipairs(C.PARTS) do byId[p.id] = p; p.price = p.price or 0 end
for _, c in ipairs(C.CATS) do catById[c.id] = c end
function C.part(id) return byId[id] end
function C.cat(id) return catById[id] end
function C.partsIn(cat) local out = {}; for _, p in ipairs(C.PARTS) do if p.cat == cat then out[#out + 1] = p end end return out end

-- a part is owned when it is free, bought with points, or its Robux pass is owned
function C.owned(state, p)
	if p.robux then return state.passes and state.passes[p.robux] == true end
	return p.price == 0 or (state.owned and state.owned[p.id] == true)
end
-- a full equipment table from a possibly partial or invalid one (unknown or unowned parts fall back to stock)
function C.clean(state, eq)
	local out = {}
	for k, v in pairs(C.DEFAULT) do out[k] = v end
	if type(eq) == "table" then
		for _, cat in ipairs(C.CATS) do
			local id = eq[cat.id]
			local p = type(id) == "string" and byId[id]
			if p and p.cat == cat.id and C.owned(state, p) then out[cat.id] = id end
		end
		if type(eq.col) == "number" and eq.col >= 1 and eq.col <= #C.COLOURS and eq.col % 1 == 0 then out.col = eq.col end
	end
	return out
end
-- base car + equipped performance parts -> tuned stats (top in m/s, power in W). Callers finish drag and gearing themselves.
function C.tune(base, eq)
	local c = {}
	for k, v in pairs(base) do c[k] = v end
	local top, power, grip, brake, wet, soft = 0, 0, 0, 0, nil, 0
	for _, cat in ipairs(C.CATS) do
		if cat.perf then
			local p = byId[(eq and eq[cat.id]) or C.DEFAULT[cat.id]]
			local fx = p and p.fx or {}
			top += (fx.topMph or 0) / 2.23694; power += fx.power or 0; grip += fx.grip or 0; brake += fx.brake or 0
			wet = fx.wet or wet; soft = math.max(soft, fx.landSoft or 0)
		end
	end
	c.top = base.top + top; c.power = base.power * (1 + power); c.grip = base.grip * (1 + grip); c.brake = base.brake * (1 + brake)
	c.wet = wet or 0.85; c.landSoft = soft
	return c
end
return C
